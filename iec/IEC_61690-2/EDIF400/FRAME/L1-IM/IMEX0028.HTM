pcb_part_model

Description:

The pcb_part_model schema is used to describe parts. This information changes infrequently and is used for the design of many boards. A component is an instance of a part. For example, there may be three 74LS00 components in an assembly. However, they share the same characteristics which are defined by a 74LS00 part.

part

  ENTITY part
      ABSTRACT SUPERTYPE OF (ONEOF (electrical_part,
                                    mechanical_part,
                                    connector_part,
                                    jumper_part));
    document         : OPTIONAL SET [1:?] OF documentation;
    name             : OPTIONAL name_information;
    previous_version : OPTIONAL part; -- reference
    properties       : OPTIONAL SET [1:?] OF property;
    units            : part_units;
    written_status   : OPTIONAL SET [1:?] OF written;
  WHERE
    valid_previous_version :
      (* A part cannot be a new version of itself, either directly or
         indirectly. *)
      non_recursive_previous_version(SELF, []);

    valid_written_in_previous_version :
      (* The earliest timestamp of a part is later than the latest
         timestamp of its previous version. *)  
      valid_timestamp_in_previous_version(written_status,
                                          previous_version.written_status);
  END_ENTITY;

Description:

A part is the set of information which describes a particular component. It corresponds to a data book description of a component. A part may be either electrical_part, mechanical_part, connector_part or jumper_part.

die_electrical_part

  ENTITY die_electrical_part
      SUBTYPE OF (electrical_part);
    SELF\electrical_part.terminals   : SET [1:?] OF
                                       die_electrical_part_terminal; 
    SELF\part.previous_version       : OPTIONAL die_electrical_part; -- retype
    referenced_die_physical_template : -- reference
                                       OPTIONAL die_physical_template;
  WHERE

      (* If it is an external_electrical_part, it is not allowed to have any
         template information. *)
    NOT (is_typeof(SELF, 'pcb_part_model.external_electrical_part')
         AND
         EXISTS(referenced_die_physical_template));
  END_ENTITY;

Description:

A die_electrical_part is the set of information which describes a die electrical component. A die electrical component is unpackaged, pre-fabricated and contains some electrical functionality.

The count of die_electrical_part_terminals defined within a die_electrical_part may be less than the count of bond_pads defined within the die_physical_template that is referenced by the die_electrical_part. This means that it is possible for some of the bond_pads of the die_physical_template to be referenced by none of the die_electrical_part_terminals. This does not mean that those bond_pads are unconnected in the die_electrical_part; it means only that the die_electrical_part is not completely defined.

An unconnected bond_pad is indicated in a die_electrical_part by defining the corresponding fully_defined_electrical_part_terminal as having a reference to a bond_pad, but no references to any part_function_terminals.

Used by:

die_electrical_component die_electrical_part die_electrical_part_terminal embedded_die

  ENTITY fully_defined_electrical_part
      SUBTYPE OF (electrical_part);
    SELF\electrical_part.terminals 
        : SET [1:?] OF fully_defined_electrical_part_terminal;
    functions            : SET [1:?] OF part_function;
    permutable_functions : OPTIONAL SET [1:?] OF
                           permutable_part_function_relationship; 
  DERIVE
    terminal_permutability : SET OF permutable_part_terminal_relationship
      := derive_part_terminal_permutability(functions, 
                                            permutable_functions);

    valid_functions :

         to external_clusters only. *)

           AND
           there_exists(QUERY(fn <* functions |
             is_typeof(fn.instantiated_cluster, 
                       'hierarchy_model.internal_cluster'))));

part_function_terminal

  ENTITY part_function_terminal;
    properties               : OPTIONAL SET [1:?] OF property;
    referenced_part_function : part_function; -- reference
    referenced_terminal      : master_port;   -- reference
  UNIQUE
    uniqueness_constraint :
      (* There is only one part_function_terminal for a particular master_port
         on a part_function. *)
      referenced_part_function, referenced_terminal;
  WHERE
    valid_referenced_terminal :
      (* The referenced master_port must be defined in the interface of the
         instantiated cluster of the referenced part_function. *)
      EXISTS(referenced_part_function.instantiated_cluster.interface.ports)
      AND
      (referenced_terminal 
         IN referenced_part_function.instantiated_cluster.interface.ports);
  END_ENTITY;

  ENTITY ibis_part_characteristic
      SUBTYPE OF (part_characteristic);
    ibis_file                     : ibis_file_information;
    ibis_component_identification : string_token;
    ibis_pin_bindings             : SET [1:?] OF ibis_pin_binding;
  DERIVE
    containing_electrical_part : internal_electrical_part -- reference
      := usedin_one_role (SELF,
           'pcb_part_model.internal_electrical_part.part_information');
    containing_connector_part : internal_connector_part -- reference
      := usedin_one_role (SELF,
           'pcb_part_model.internal_connector_part.part_information');
    containing_jumper_part : internal_jumper_part -- reference
      := usedin_one_role (SELF,
           'pcb_part_model.internal_jumper_part.part_information');
  WHERE
    valid_component_identification :
      (* The ibis_component_identification is a maximum of 40 characters. *)
      LENGTH (ibis_component_identification) <= 40;

    valid_pin_bindings :
      (* Each terminal of the part has a corresponding ibis_pin_binding. *)
      (EXISTS(containing_electrical_part) AND
        (SIZEOF(ibis_pin_bindings) 
         = SIZEOF(containing_electrical_part.terminals)))
      OR
      (EXISTS(containing_connector_part) AND
        (SIZEOF(ibis_pin_bindings) 
         = SIZEOF(containing_connector_part.terminals)))
      OR
      (EXISTS(containing_jumper_part) AND
        (SIZEOF(ibis_pin_bindings) 
         = SIZEOF(containing_jumper_part.terminals)));
  END_ENTITY;

ibis_pin_binding

  ENTITY ibis_pin_binding;
    part_terminal           : part_terminal; -- reference
    ibis_pin_identification : string_token;
    properties              : OPTIONAL SET [1:?] OF property;
  DERIVE
    containing_ibis_information : ibis_part_characteristic -- reference
      := usedin_one_role(SELF,
         'pcb_part_model.ibis_part_characteristic.ibis_pin_bindings');
  UNIQUE
    unique_part_terminal :
      (* There is only one IBIS correspondent for a given part_terminal. *)
      part_terminal;

    unique_ibis_pin :
      (* Within an IBIS component, an IBIS pin can only correspond to
         one part_terminal. *)
      ibis_pin_identification, containing_ibis_information;
  WHERE
    valid_existence :
      (* An ibis_pin_binding belongs to one ibis_part_characteristic. *)
      belongs_to_one (SELF,
         'pcb_part_model.ibis_part_characteristic.ibis_pin_bindings');

    valid_part_terminal :
      (* The part_terminal is a terminal of the part being mapped. *)
      (is_typeof(part_terminal, 'pcb_part_model.electrical_part_terminal')
       AND
       EXISTS(containing_ibis_information.containing_electrical_part)
       AND
       (part_terminal IN 
        containing_ibis_information.containing_electrical_part.terminals))
      OR
      (is_typeof(part_terminal, 'pcb_part_model.connector_part_terminal')
       AND
       EXISTS(containing_ibis_information.containing_connector_part)
       AND
       (part_terminal IN 
        containing_ibis_information.containing_connector_part.terminals))
      OR
      (is_typeof(part_terminal, 'pcb_part_model.jumper_part_terminal')
       AND
       EXISTS(containing_ibis_information.containing_jumper_part)
       AND
       (part_terminal IN 
        containing_ibis_information.containing_jumper_part.terminals));

    valid_pin_identification :
     (* The ibis_pin_identification is a maximum of 5 characters. *)
     LENGTH (ibis_pin_identification) <= 5;
  END_ENTITY;
