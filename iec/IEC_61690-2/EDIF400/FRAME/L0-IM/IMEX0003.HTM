design_hierarchy_model

Description:

The design_hierarchy_model schema describes the annotation data relating to an occurrence hierarchy. A design_hierarchy is defined by choosing the top cluster_configuration. It is annotated by an occurrence_hierarchy_annotate. Interconnects and master ports within the top occurrence can be annotated by interconnect_annotate and master_port_annotate respectively. If further annotation is to be given for occurrences further down in the occurrence hierarchy, occurrence_annotate can be used to specify which occurrence is to be annotated next. Similarly, interconnects and instance ports within an occurrence can be annotated by interconnect_annotate and instance_port_annotate respectively.

       EXISTS(top_cell.clusters) OR NOT EXISTS(design_hierarchies);

occurrence_hierarchy_annotate

  ENTITY occurrence_hierarchy_annotate;
    interconnect_annotates       : OPTIONAL SET [1:?] OF interconnect_annotate;
    leaf_occurrence_annotates    : OPTIONAL SET [1:?] OF
                                   leaf_occurrence_annotate; 
    master_port_annotates        : OPTIONAL SET [1:?] OF master_port_annotate;
    occurrence_annotates         : OPTIONAL SET [1:?] OF occurrence_annotate;
    overriding_cell_properties   : OPTIONAL SET [1:?] OF property_override;
    overriding_cluster_properties: OPTIONAL SET [1:?] OF property_override;
    overriding_designator        : OPTIONAL string_token;
    overriding_view_properties   : OPTIONAL SET [1:?] OF property_override;
    signal_annotates             : OPTIONAL SET [1:?] OF signal_annotate;
  DERIVE
    containing_design_hierarchy : design_hierarchy -- reference
    := usedin_one_role(SELF,
       'design_hierarchy_model.design_hierarchy.occurrence_hierarchy_annotate');
  WHERE
    containment_constraint :
      (* An occurrence_hierarchy_annotate belongs to one design_hierarchy. *)
      belongs_to_one(SELF,
      'design_hierarchy_model.design_hierarchy.occurrence_hierarchy_annotate');

    valid_interconnect_annotates :
      (* The annotated interconnects are defined in the chosen_view of the
         top occurrence.  No two interconnect_annotates annotate the same
         interconnect. *)
      valid_interconnect_annotate(interconnect_annotates,
        containing_design_hierarchy.chosen_cluster_configuration)
      AND
      not_exists(QUERY(an1 <* interconnect_annotates |
        there_exists(QUERY(an2 <* interconnect_annotates - an1 |
          an1.annotated_interconnect :=: an2.annotated_interconnect))));

    valid_leaf_occurrence_annotates :
      (* If the chosen cluster_configuration is an
         expandable_cluster_configuration, the annotated leaf occurrences must
         reference instance_configurations which are defined in the chosen
         cluster_configuration of the containing design_hierarchy. *)
      NOT 
      (is_typeof(containing_design_hierarchy.chosen_cluster_configuration,
                 'hierarchy_model.expandable_cluster_configuration')
       AND
       there_exists(QUERY(leafOccurrenceAnnotate <* leaf_occurrence_annotates |
         NOT (leafOccurrenceAnnotate.chosen_instance_configuration
              IN containing_design_hierarchy
                 .chosen_cluster_configuration\expandable_cluster_configuration
                 .instance_configurations))));

    valid_master_port_annotates :
      (* The annotated master_ports are defined in the interface of the
         containing cluster of the chosen_configuration of the containing
         design_hierarchy.  No two master_port_annotates annotate the same
         master_port. *)   
      not_exists(QUERY(portAnnotate <* master_port_annotates |
        NOT (portAnnotate.annotated_port 
             IN containing_design_hierarchy.chosen_cluster_configuration
                .containing_cluster.interface.ports)))
      AND
      not_exists(QUERY(annotate1 <* master_port_annotates |
        there_exists(QUERY(annotate2 <* master_port_annotates - annotate1 |
           annotate1.annotated_port :=: annotate2.annotated_port))));

    valid_occurrence_annotates :
      (* If the chosen cluster_configuration is an
         expandable_cluster_configuration, the next level occurrence_annotates 
         must reference instance_configurations which are defined in the chosen
         cluster_configuration of the containing design_hierarchy. *) 
      NOT
      (is_typeof(containing_design_hierarchy.chosen_cluster_configuration,
                 'hierarchy_model.expandable_cluster_configuration')
       AND
       there_exists(QUERY(occurrenceAnnotate <* occurrence_annotates |
         NOT (occurrenceAnnotate.chosen_instance_configuration
              IN containing_design_hierarchy
                 .chosen_cluster_configuration\expandable_cluster_configuration
                 .instance_configurations))));

    valid_overriding_cell_properties :
      (* The overridden properties are defined in the containing cell of the
         chosen cluster_configuration of the containing design_hierarchy. *)
      not_exists(QUERY(cellPropertyOverride <* overriding_cell_properties |
        NOT (cellPropertyOverride.overridden_property
             IN containing_design_hierarchy.chosen_cluster_configuration
                .containing_cluster.containing_cell.properties)));

    valid_overriding_cluster_properties :
      (* The overridden properties are defined in the containing cluster of the
         chosen cluster_configuration of the containing design_hierarchy. *)
     not_exists(QUERY(clusterPropertyOverride <* overriding_cluster_properties |
        NOT (clusterPropertyOverride.overridden_property
             IN containing_design_hierarchy.chosen_cluster_configuration
                .containing_cluster.properties)));

    valid_overriding_view_properties :
      (* If the chosen cluster_configuration is an
         expandable_cluster_configuration, the overridden properties are
         defined in the chosen cell_representation of the chosen
         cluster_configuration of the containing design_hierarchy. *) 
      NOT
      (is_typeof(containing_design_hierarchy.chosen_cluster_configuration,
                 'hierarchy_model.expandable_cluster_configuration')
       AND
       there_exists(QUERY(viewPropertyOverride <* overriding_view_properties |
         NOT (viewPropertyOverride.overridden_property 
              IN containing_design_hierarchy
                 .chosen_cluster_configuration\expandable_cluster_configuration
                 .chosen_cell_representation.properties))));

    valid_signal_annotates :
      (* The annotated signals are defined in the chosen_view of the top
         occurrence.  No two signal_annotates annotate the same signal. *)  
      NOT
      (is_typeof(containing_design_hierarchy.chosen_cluster_configuration,
                 'hierarchy_model.expandable_cluster_configuration')
       AND
       there_exists(QUERY(signalAnnotate <* signal_annotates |
         signalAnnotate.annotated_signal.containing_view
         :<>: containing_design_hierarchy.chosen_cluster_configuration
              \expandable_cluster_configuration.chosen_cell_representation)))
      AND
      not_exists(QUERY(an1 <* signal_annotates |
        there_exists(QUERY(an2 <* signal_annotates - an1 |
          an1.annotated_signal :=: an2.annotated_signal))));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. An occurrence_hierarchy_annotate cannot have
            interconnect_annotates, leaf_occurrence_annotates,
            occurrence_annotates, overriding_view_properties, signal_annotates
            if the chosen cluster_configuration is a leaf_configuration. 
         2. An occurrence_hierarchy_annotate cannot have
            leaf_occurrence_annotates or occurrence_annotates if the chosen
            expandable_cluster_configuration does not define any
            instance_configurations. 
         3. An occurrence_hierarchy_annotate cannot have master_port_annotates
            if the interface of the annotated cluster does not define any
            ports. 
         4. An occurrence_hierarchy_annotate cannot have
            overriding_cell_properties if the annotated cell does not define
            any properties. 
         5. An occurrence_hierarchy_annotate cannot have
            overriding_cluster_properties if the annotated cluster does not
            define any properties.
         6. An occurrence_hierarchy_annotate cannot have
            overriding_view_properties if the annotated view does not
            define any properties. *)

      NOT (is_typeof(containing_design_hierarchy.chosen_cluster_configuration,
                     'hierarchy_model.leaf_configuration') AND
           EXISTS(interconnect_annotates) OR 
           EXISTS(leaf_occurrence_annotates) OR
           EXISTS(occurrence_annotates) OR
           EXISTS(overriding_view_properties) OR
           EXISTS(signal_annotates))
      AND
      ((is_typeof(containing_design_hierarchy.chosen_cluster_configuration,
                  'hierarchy_model.expandable_cluster_configuration') AND
        EXISTS(containing_design_hierarchy.chosen_cluster_configuration
               \expandable_cluster_configuration.instance_configurations)) OR
          NOT EXISTS(leaf_occurrence_annotates) AND
          NOT EXISTS(occurrence_annotates))
      AND
      (EXISTS(containing_design_hierarchy.chosen_cluster_configuration
              .containing_cluster.interface.ports) OR
         NOT EXISTS(master_port_annotates))
      AND
      (EXISTS(containing_design_hierarchy.chosen_cluster_configuration
              .containing_cluster.containing_cell.properties) OR
         NOT EXISTS(overriding_cell_properties))
      AND
      (EXISTS(containing_design_hierarchy.chosen_cluster_configuration
              .containing_cluster.properties) OR
         NOT EXISTS(overriding_cluster_properties))
      AND
      ((is_typeof(containing_design_hierarchy.chosen_cluster_configuration,
                  'hierarchy_model.expandable_cluster_configuration') AND
        EXISTS(containing_design_hierarchy.chosen_cluster_configuration
               \expandable_cluster_configuration.chosen_cell_representation
               .properties)) OR
        NOT EXISTS(overriding_view_properties));
  END_ENTITY;

occurrence_annotate

  ENTITY occurrence_annotate;
   chosen_instance_configuration : instance_configuration; -- reference
   instance_member_index         : integer_token;
   instance_port_annotates       : OPTIONAL SET [1:?] OF instance_port_annotate;
   interconnect_annotates        : OPTIONAL SET [1:?] OF interconnect_annotate;
   leaf_occurrence_annotates     : OPTIONAL SET [1:?] OF
                                   leaf_occurrence_annotate; 
   new_instance_properties       : OPTIONAL SET [1:?] OF property;
   occurrence_annotates          : OPTIONAL SET [1:?] OF occurrence_annotate;
   overriding_cell_properties    : OPTIONAL SET [1:?] OF property_override;
   overriding_cluster_properties : OPTIONAL SET [1:?] OF property_override;
   overriding_designator         : OPTIONAL string_token;
   overriding_instance_properties: OPTIONAL SET [1:?] OF property_override;
   overriding_view_properties    : OPTIONAL SET [1:?] OF property_override;
   signal_annotates              : OPTIONAL SET [1:?] OF signal_annotate;
  DERIVE
    annotated_instance : instance -- reference
      := chosen_instance_configuration.configured_instance;
    annotated_view : cell_representation -- reference
      := chosen_instance_configuration.chosen_cluster_configuration
         \expandable_cluster_configuration.chosen_cell_representation; 
  WHERE
    containment_constraint :
      (* An occurrence_annotate belongs to either one
         occurrence_hierarchy_annotate or one occurrence_annotate . *)
   either_belongs_to_one(SELF,
   ['design_hierarchy_model.occurrence_hierarchy_annotate.occurrence_annotates',
    'design_hierarchy_model.occurrence_annotate.occurrence_annotates']);

    valid_chosen_instance_configuration :
      (* The chosen instance_configuration must reference an
         expandable_cluster_configuration. *) 
      is_typeof(chosen_instance_configuration.chosen_cluster_configuration,
                'hierarchy_model.expandable_cluster_configuration');

    valid_instance_member_index :
      (* The member index is within the range defined by the width of the
         annotated instance. *) 
      {0 <= instance_member_index < annotated_instance.width};

    valid_instance_port_annotates :
      (* The annotated instance_ports reference the annotated_instance.  No two
         instance_port_annotates annotate the same instance_port. *) 
        not_exists(QUERY(portAnnotate <* instance_port_annotates |
          NOT (portAnnotate.annotated_port.referenced_instance 
               :<>: annotated_instance)))
    AND not_exists(QUERY(portAnnotate1 <* instance_port_annotates |
          there_exists(QUERY(portAnnotate2 
            <* instance_port_annotates - portAnnotate1 |
            portAnnotate1.annotated_port :=: portAnnotate2.annotated_port))));

    valid_interconnect_annotates :
      (* The annotated interconnects are defined in the chosen_view for the
         annotated_instance.  No two interconnect_annotates annotate the same
         interconnect. *)
      valid_interconnect_annotate(interconnect_annotates,
        chosen_instance_configuration.chosen_cluster_configuration)
      AND
      not_exists(QUERY(an1 <* interconnect_annotates |
        there_exists(QUERY(an2 <* interconnect_annotates - an1 |
          an1.annotated_interconnect :=: an2.annotated_interconnect))));

    valid_leaf_occurrence_annotates :
      (* The annotated leaf occurrences must reference instance_configurations
         defined in the cluster_configuration which are chosen by the
         instance_configuration. *)
      not_exists(QUERY(leafOccurrenceAnnotate <* leaf_occurrence_annotates |
        NOT (leafOccurrenceAnnotate.chosen_instance_configuration
             IN chosen_instance_configuration.chosen_cluster_configuration
                \expandable_cluster_configuration.instance_configurations)));

    valid_occurrence_annotates :
      (* The next level occurrence_annotates must reference
         instance_configurations defined in the cluster_configuration which are
         chosen by the instance_configuration. *) 
      not_exists(QUERY(occurrenceAnnotate <* occurrence_annotates |
        NOT (occurrenceAnnotate.chosen_instance_configuration
             IN chosen_instance_configuration.chosen_cluster_configuration
                \expandable_cluster_configuration.instance_configurations)));

    valid_overriding_cell_properties :
      (* The overridden properties are defined in the containing cell of the
         instantiated cluster of the annotated_instance. *)
      not_exists(QUERY(cellPropertyOverride <* overriding_cell_properties |
        NOT (cellPropertyOverride.overridden_property
        IN annotated_instance.instantiated_cluster
           .containing_cell.properties)));

    valid_overriding_cluster_properties :
      (* The overridden properties are defined in the instantiated cluster of
         the annotated_instance. *)
    not_exists(QUERY(clusterPropertyOverride <* overriding_cluster_properties |
        NOT (clusterPropertyOverride.overridden_property
             IN annotated_instance.instantiated_cluster.properties)));

    valid_overriding_instance_properties :
      (* The overridden properties are defined in the annotated_instance. *)
    not_exists(QUERY(instancePropertyOverride <* overriding_instance_properties|
        NOT (instancePropertyOverride.overridden_property 
             IN annotated_instance.properties)));

    valid_overriding_view_properties :
      (* The overridden properties are defined in the chosen_view for the
         annotated_instance. *)
      not_exists(QUERY(viewPropertyOverride <* overriding_view_properties |
        NOT (viewPropertyOverride.overridden_property 
             IN annotated_view.properties)));

    valid_signal_annotates :
      (* The annotated signals are defined in the chosen_view for the
         annotated_instance.  No two signal_annotates annotate the same
         signal. *)
       not_exists(QUERY(signalAnnotate <* signal_annotates |
         signalAnnotate.annotated_signal.containing_view :<>: annotated_view))
       AND
        not_exists(QUERY(an1 <* signal_annotates |
          there_exists(QUERY(an2 <* signal_annotates - an1 |
            an1.annotated_signal :=: an2.annotated_signal))));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. An occurrence_annotate cannot have leaf_occurrence_annotates or
            occurrence_annotates if the chosen expandable_cluster_configuration
            does not define any instance_configurations. 
         2. An occurrence_annotate cannot have overriding_cell_properties if
            the annotated cell does not define any properties.
         3. An occurrence_annotate cannot have overriding_cluster_properties if
            the annotated cluster does not define any properties.
         4. An occurrence_annotate cannot have overriding_instance_properties
            if the annotated instance does not define any properties.
         5. An occurrence_annotate cannot have overriding_view_properties
            if the annotated view does not define any properties. *)
    (EXISTS(chosen_instance_configuration.chosen_cluster_configuration
            \expandable_cluster_configuration.instance_configurations) OR
       NOT EXISTS(leaf_occurrence_annotates) AND
       NOT EXISTS(occurrence_annotates))
    AND
    (EXISTS(annotated_instance.instantiated_cluster.containing_cell.properties)
       OR NOT EXISTS(overriding_cell_properties))
    AND
    (EXISTS(annotated_instance.instantiated_cluster.properties) OR
       NOT EXISTS(overriding_cluster_properties))
    AND
    (EXISTS(annotated_instance.properties) OR
       NOT EXISTS(overriding_instance_properties))
    AND
    (EXISTS(annotated_view.properties) OR
       NOT EXISTS(overriding_view_properties));
  END_ENTITY;

  ENTITY master_port_annotate

        NOT (delayOverride.overridden_delay IN annotated_port.delays)));

      not_exists(QUERY(delayOverride <* overriding_load_delays |

             IN annotated_port.load_delays)));

      (* The overridden properties are defined in the annotated master_port. *)
      not_exists(QUERY(portPropertyOverride <* overriding_properties |
        NOT (portPropertyOverride.overridden_property
             IN annotated_port.properties)));

            annotated port does not define any load delays.

      (EXISTS(annotated_port.load_delays) OR NOT EXISTS(overriding_load_delays))
      AND
      (EXISTS(annotated_port.properties) OR NOT EXISTS(overriding_properties));

  ENTITY instance_port_annotate
    ABSTRACT SUPERTYPE OF (ONEOF(input_instance_port_annotate, 
                                 output_instance_port_annotate,
                                 bidirectional_instance_port_annotate, 
                                 unspecified_direction_instance_port_annotate));
    annotated_port         : instance_port; -- reference
    new_delays             : OPTIONAL SET [1:?] OF port_delay;
    new_load_delays        : OPTIONAL SET [1:?] OF port_load_delay;
    new_properties         : OPTIONAL SET [1:?] OF property;
    overriding_ac_load     : OPTIONAL capacitance_value;
    overriding_delays      : OPTIONAL SET [1:?] OF port_delay_override;
    overriding_designator  : OPTIONAL string_token;
    overriding_load_delays : OPTIONAL SET [1:?] OF port_load_delay_override;
    overriding_properties  : OPTIONAL SET [1:?] OF property_override;
  WHERE
    containment_constraint :
      (* An instance_port_annotate belongs to either one occurrence_annotate or
         one leaf_occurrence_annotate. *) 
   either_belongs_to_one(SELF,
   ['design_hierarchy_model.occurrence_annotate.instance_port_annotates',
    'design_hierarchy_model.leaf_occurrence_annotate.instance_port_annotates']);

    valid_overriding_delays :
      (* The overridden delays are defined in the annotated instance_port or in
         the referenced master_port definition. *)
      not_exists(QUERY(delayOverride <* overriding_delays |
        NOT ((delayOverride.overridden_delay IN annotated_port.new_delays)
              OR (delayOverride.overridden_delay 
                  IN annotated_port.referenced_master_port.delays))));

    valid_overriding_load_delays :
      (* The overridden load_delays are defined in the annotated instance_port
         or in the referenced master_port definition. *)
      not_exists(QUERY(delayOverride <* overriding_load_delays |
        NOT ((delayOverride.overridden_load_delay 
              IN annotated_port.new_load_delays)
              OR (delayOverride.overridden_load_delay 
                  IN annotated_port.referenced_master_port.load_delays))));

    valid_overriding_properties :
      (* The overridden properties are defined in the annotated_port or in the
         referenced master_port definition. *)
      not_exists(QUERY(portPropertyOverride <* overriding_properties |
        NOT ((portPropertyOverride.overridden_property
              IN annotated_port.new_properties)
              OR (portPropertyOverride.overridden_property
                  IN annotated_port.referenced_master_port.properties))));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. An instance_port_annotate cannot have overriding_delays if its
            annotated port does not define any delays.
         2. An instance_port_annotate cannot have overriding_load_delays if its
            annotated port does not define any load delays.
         3. An instance_port_annotate cannot have overriding_properties if its
            annotated port does not define any properties. *)
      (EXISTS(annotated_port.referenced_master_port.delays) OR 
         NOT EXISTS(overriding_delays))
      AND
      (EXISTS(annotated_port.referenced_master_port.load_delays) OR 
         NOT EXISTS(overriding_load_delays))
      AND
      (EXISTS(annotated_port.referenced_master_port.properties) OR 
         NOT EXISTS(overriding_properties));
  END_ENTITY;

valid_interconnect_annotate

  FUNCTION valid_interconnect_annotate
    (annotates : SET OF interconnect_annotate;
     config    : cluster_configuration) : BOOLEAN;
    LOCAL
      annotatedView         : cell_representation;
      annotatedInterconnect : referenced_interconnect;
    END_LOCAL;

    IF is_typeof(config,
       'hierarchy_model.expandable_cluster_configuration') THEN
       annotatedView := config\expandable_cluster_configuration
                              .chosen_cell_representation;
       IF is_typeof(annotatedView, 
          'connectivity_view_model.internal_connectivity_view') THEN
          REPEAT i := 1 TO SIZEOF(annotates);
            annotatedInterconnect := annotates[i].annotated_interconnect;
            IF is_typeof(annotatedInterconnect,
               'connectivity_structure_model.connectivity_generic_bus') THEN
               IF containing_view_of_connectivity_bus_structure
                  (annotatedInterconnect) :<>: annotatedView THEN
                  RETURN(FALSE);
               END_IF;
            ELSE
            IF is_typeof(annotatedInterconnect,
               'connectivity_structure_model.connectivity_generic_net') THEN
               IF containing_view_of_connectivity_net_structure
                  (annotatedInterconnect) :<>: annotatedView THEN
                  RETURN(FALSE);
               END_IF;
            END_IF;
            END_IF;
          END_REPEAT;
       ELSE
       IF is_typeof(annotatedView,
          'schematic_view_model.internal_schematic_view') THEN
          REPEAT i := 1 TO SIZEOF(annotates);
            annotatedInterconnect := annotates[i].annotated_interconnect;
            IF is_typeof(annotatedInterconnect,
               'schematic_implementation_model.schematic_generic_bus') THEN
               IF containing_bus_of_schematic_bus_structure
                  (annotatedInterconnect).containing_page.containing_view
                  :<>: annotatedView THEN
                  RETURN(FALSE);
               END_IF;
            ELSE
            IF is_typeof(annotatedInterconnect,
               'schematic_implementation_model.schematic_generic_net') THEN
               IF containing_net_of_schematic_net_structure
                  (annotatedInterconnect).containing_page.containing_view
                  :<>: annotatedView THEN
                  RETURN(FALSE);
               END_IF;
            END_IF;
            END_IF;
          END_REPEAT;
       END_IF;
       END_IF;
    END_IF;
  END_FUNCTION;

The valid_interconnect_annotate function returns true if all annotated interconnects are defined in the chosen view of the cluster_configuration.
