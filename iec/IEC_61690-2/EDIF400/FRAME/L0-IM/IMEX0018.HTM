physical_figure

  ENTITY physical_figure;
    associated_geometries      : OPTIONAL SET [1:?] OF geometry_element;
    associated_geometry_macros : OPTIONAL SET [1:?] OF geometry_macro;
    default_attributes         : pcb_attribute_default; -- reference
    overriding_colour          : OPTIONAL colour;
    overriding_corner_type     : OPTIONAL path_corner_type;
    overriding_end_type        : OPTIONAL path_end_type;
    overriding_path_width      : OPTIONAL length_value;
    overriding_properties      : OPTIONAL SET [1:?] OF property_override;
  WHERE
    valid_overriding_properties :
      (* The overridden properties are defined in the pcb_attribute_default.*)
      not_exists(QUERY(propertyOverride <* overriding_properties |
        NOT (propertyOverride.overridden_property
             IN default_attributes\geometric_attribute_default.properties)));

    valid_associated_geometries :
      (* The associated geometry_elements must not contain schematic specific
         geometry objects.  That is, they are not allowed to have elements
         containing dot. *)   
      not_exists(QUERY(geometryElement <* associated_geometries |
        is_typeof(geometryElement, 'geometry_model.dot')));

    valid_associated_geometry_macros :
      (* The associated geometry_macros must not reference
         geometry_macro_templates which contain PCB specific geometry objects.
         That is, the referenced geometry_macro_templates must not include
         elements containing dot. *)
      valid_geometry_macro_in_physical_figure(associated_geometry_macros);
  END_ENTITY;

valid_geometry_macro_in_physical_figure

  FUNCTION valid_geometry_macro_in_physical_figure
    (elements : SET OF geometry_macro) : BOOLEAN;

    REPEAT i := 1 TO SIZEOF(elements);
      IF is_typeof(elements[i].instantiated_template,
         'instantiatable_object_model.internal_geometry_macro_template') THEN
         IF there_exists(QUERY(element 
                         <* elements[i].instantiated_template
                            \internal_geometry_macro_template.geometries |
                         is_typeof(element, 'geometry_model.dot')))
            OR
            NOT valid_geometry_macro_in_physical_figure(
                   elements[i].instantiated_template
                   \internal_geometry_macro_template.geometry_macros) THEN
            RETURN(FALSE);
      END_IF;
    END_IF;
    END_REPEAT;    
    RETURN(TRUE);
  END_FUNCTION;

Description:

The valid_geometry_macro_in_physical_figure function is used to check that all geometry_macros within a physical_figure must not include elements containing dot.

point_inside_fabricate_figure

  FUNCTION point_inside_fabricate_figure
    (pt : point; fig : fabricate_figure) : BOOLEAN;
     REPEAT i := 1 TO SIZEOF(fig.associated_figures);
       IF point_inside_physical_figure(pt, fig.associated_figures[i]) THEN
          RETURN(TRUE);
       END_IF;
     END_REPEAT;
     REPEAT i := 1 TO SIZEOF(fig.associated_figure_macros);
       IF point_inside_physical_figure_macro(pt, 
                                         fig.associated_figure_macros[i]) THEN
          RETURN(TRUE);
       END_IF;
     END_REPEAT;
     RETURN(FALSE);
  END_FUNCTION;

Description:

The point_inside_fabricate_figure returns true if a point lies inside a fabricate_figure.

point_inside_physical_figure

  FUNCTION point_inside_physical_figure
    (pt : point; element : physical_figure) : BOOLEAN;
     REPEAT i := 1 TO SIZEOF(element.associated_geometries);
       IF point_inside_geometry_element(pt, 
          element.associated_geometries[i]) THEN
          RETURN(TRUE);
       END_IF;
     END_REPEAT;
     REPEAT i := 1 TO SIZEOF(element.associated_geometry_macros);
       IF point_inside_geometry_macro(pt, 
          element.associated_geometry_macros[i]) THEN
          RETURN(TRUE);
       END_IF;
     END_REPEAT;
     RETURN(FALSE);
  END_FUNCTION;

Description:

The point_inside_physical_figure returns true if a point lies inside a physical_figure.

point_inside_physical_figure_macro

  FUNCTION point_inside_physical_figure_macro
    (pt : point; element : physical_figure_macro) : BOOLEAN;
  IF is_typeof(element.instantiated_template,
  'pcb_mcm_instantiatable_object_model.internal_physical_figure_macro_template')
     THEN
        IF there_exists(QUERY(geometry 
                              <* element.instantiated_template
                                        \internal_physical_figure_macro_template
                                        .associated_figures |
                        point_inside_physical_figure(pt, geometry))) THEN
           RETURN(TRUE);
        END_IF;
        IF there_exists(QUERY(macro
                              <* element.instantiated_template
                                        \internal_physical_figure_macro_template
                                        .associated_figure_macros |
                        point_inside_physical_figure_macro(pt, macro))) THEN
           RETURN(TRUE);
        END_IF;
  ELSE
        RETURN(TRUE);
  END_IF;
  RETURN(FALSE);
  END_FUNCTION;

The point_inside_physical_figure_macro returns true if a point lies inside a physical_figure_macro.
