        NOT (figureGroupPropertyOverride.overridden_property
             IN associated_figure_group.properties)));

         geometry objects.  That is, they are not allowed to have elements
         containing arc_by_centre, circle_by_centre,
         closed_area_element_with_cutout or unfilled_element. *)

valid_geometry_element_in_figure

  FUNCTION valid_geometry_element_in_figure
    (elements : SET OF geometry_element) : BOOLEAN;
    LOCAL
      elementType : SET OF STRING;
    END_LOCAL;

    REPEAT i := 1 TO SIZEOF(elements);
      elementType := TYPEOF(elements[i]);
      IF there_exists(['geometry_model.circle_by_centre',
                       'geometry_model.closed_area_element_with_cutout',
                       'geometry_model.unfilled_element'] * elementType) THEN
         RETURN(FALSE);
      ELSE 
      IF  'geometry_model.shape' IN elementType THEN 
          IF there_exists(QUERY(segment <* elements[i]\shape.segments |
             'geometry_model.arc_by_centre' IN TYPEOF(segment))) THEN
             RETURN(FALSE);
          END_IF;
      ELSE 
      IF  'geometry_model.open_shape' IN elementType THEN 
           IF there_exists(QUERY(segment <* elements[i]\open_shape.segments |
              'geometry_model.arc_by_centre' IN TYPEOF(segment))) THEN
              RETURN(FALSE);
          END_IF;
      ELSE 
      IF  'geometry_model.unfilled_shape' IN elementType THEN 
          IF there_exists(QUERY(segment <* elements[i]\unfilled_shape.segments |
             'geometry_model.arc_by_centre' IN TYPEOF(segment))) THEN
             RETURN(FALSE);
          END_IF;
      END_IF;
      END_IF;
      END_IF;
      END_IF;
    END_REPEAT;    
    RETURN(TRUE);
  END_FUNCTION;

Description:

The valid_geometry_element_in_figure function is used to check that all geometry_elements within a figure must not contain PCB specific geometry objects. The function returns true if a figure does not contain elements containing arc_by_centre, circle_by_centre, closed_area_element_with_cutout or unfilled_element.

Used by:

figure

valid_geometry_macro_in_figure

  FUNCTION valid_geometry_macro_in_figure
    (elements : SET OF geometry_macro) : BOOLEAN;
    LOCAL
      elementType : SET OF STRING;
    END_LOCAL;

    REPEAT i := 1 TO SIZEOF(elements);
      elementType := TYPEOF(elements[i].instantiated_template);
      IF 'instantiatable_object_model.internal_geometry_macro_template' 
         IN elementType THEN
         IF NOT (valid_geometry_element_in_figure(
                   elements[i].instantiated_template
                   \internal_geometry_macro_template.geometries)
                 AND
                 valid_geometry_macro_in_figure(
                   elements[i].instantiated_template
                   \internal_geometry_macro_template.geometry_macros)) THEN
            RETURN(FALSE);
      END_IF;
    END_IF;
    END_REPEAT;    
    RETURN(TRUE);
  END_FUNCTION;

Description:

The valid_geometry_macro_in_figure function is used to check that all geometry_macros within a figure must not contain PCB specific geometry objects. The function returns true if a figure does not contain elements containing arc_by_centre, circle_by_centre, closed_area_element_with_cutout or unfilled_element.
