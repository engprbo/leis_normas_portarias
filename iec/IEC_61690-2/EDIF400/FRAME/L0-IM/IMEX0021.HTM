internal_pcb_mcm_view

  ENTITY internal_pcb_mcm_view
      SUBTYPE OF (internal_cell_representation);
    components        : OPTIONAL SET [1:?] OF component;
    component_signals : OPTIONAL SET [1:?] OF component_signal;
    is_implemented_by : OPTIONAL assembled_board;
    instances         : OPTIONAL SET [1:?] OF instance;
    signals           : OPTIONAL SET [1:?] OF signal; 
    signal_groups     : OPTIONAL SET [1:?] OF signal_group; 
    swap_groups       : OPTIONAL SET [1:?] OF swap_group;
    units             : pcb_layout_units;
  WHERE
    valid_instances :
      (* An internal_pcb_mcm_view should not contain instances of clusters of
         the containing cell, either directly or indirectly. *)
      non_recursive_cell(SELF\cell_representation.containing_cluster
                             .containing_cell, instances);

    valid_instance_width :
      (* All instances must have width equal to one. *)
      not_exists(QUERY(inst <* instances |
        inst.width <> 1));

    valid_signals :
      (* All signals are disjoint.  That is only one signal may connect to a
         particular global_port, instance_member_port or master_port. *)
      not_exists(QUERY(sig1 <* signals |
        there_exists(QUERY(sig2 <* signals - sig1 |
          there_exists(sig1.joined_global_ports * sig2.joined_global_ports)
       OR there_exists(sig1.joined_instance_member_ports 
                       * sig2.joined_instance_member_ports)
       OR there_exists(sig1.joined_master_ports * sig2.joined_master_ports)))));

    valid_component_signals :
      (* All component_signals are disjoint.  That is only component_signal may
         connect to a particular component_terminal. *)
      not_exists(QUERY(sig1 <* component_signals |
        there_exists(QUERY(sig2 <* component_signals - sig1 |
          there_exists(sig1.joined_terminals * sig2.joined_terminals)))));

    valid_swap_groups :
      (* All swap_groups are disjoint.  That is only one swap_group may
         reference to a particular instance. *)
      not_exists(QUERY(swapGroup1 <* swap_groups |
        there_exists(QUERY(swapGroup2 <* swap_groups - swapGroup1 |
          there_exists(swapGroup1.instances * swapGroup2.instances)))));

    valid_optional_attributes :
       (* The dependencies of these optional attributes are stated as follows:
          1. A component_signals cannot be defined if there are no components
             defined.
          2. Signal_groups cannot be defined if there are no signals
             defined. 
          3. Swap_groups cannot be defined if there are no instances. *)
       (EXISTS(components) OR NOT EXISTS(component_signals))
       AND
       (EXISTS(signals) OR NOT EXISTS(signal_groups))
       AND
       (EXISTS(instances) OR NOT EXISTS(swap_groups));
  END_ENTITY;

Description:

An internal_pcb_mcm_view is a cell_representation which describes a printed circuit assembly. Three types of information are identified. The logical connectivity information describes the required electrical functionality that is to be realised by the assembled board, in terms of instances and signals. The component connectivity information specifies the components to be used on the assembled board and their interconnections. The physical connectivity information describes the component placements on the board and the connections implemented by conductive material.

Used by:

  ENTITY component_signal;
    joined_terminals   : SET [1:?] OF component_terminal;
    terminal_groupings : OPTIONAL SET [1:?] OF component_terminal_group;
    name               : OPTIONAL name_information;

      belongs_to_one(SELF,
        'pcb_mcm_view_model.internal_pcb_mcm_view.component_signals');

    valid_joined_terminals :
      (* All joined component_terminals refer to components in the
         containing internal_pcb_mcm_view. *)
      not_exists(QUERY(componentTerminal <* joined_terminals |
        NOT (componentTerminal.referenced_component
             IN containing_view.components)));

    valid_implementating_assembled_physical_nets :
      (* Any assembled_physical_nets which implement the
         component_signal must be in the same internal_pcb_mcm_view as the
         component_signal. *)
      not_exists(QUERY(assembledPhysicalNet <* 
                       QUERY(entityInstance <* USEDIN(SELF, '') |
                       'pcb_assembled_board_model.assembled_physical_net'
                       IN TYPEOF(entityInstance)) |
                 assembledPhysicalNet.containing_board.containing_view
                 :<>: containing_view));

    valid_implementation :
      (* A component_signal may be implemented by zero, one or more
         non_analysed_assembled_physical_nets or by exactly one
         correct_assembled_physical_net or by zero, one or more
         incorrect_assembled_physical_nets. *)
      NOT EXISTS(containing_view.is_implemented_by)
      OR
      not_exists(QUERY(entityInstance <* USEDIN(SELF, '') |
                       'pcb_assembled_board_model.assembled_physical_net'
                       IN TYPEOF(entityInstance)))
      OR
      (SIZEOF(USEDIN(SELF,
              specified_role('pcb_assembled_board_model',
                             'non_analysed_assembled_physical_net',
                             'considered_to_implement'))) >= 1)
      OR
      (SIZEOF(USEDIN(SELF,
              specified_role('pcb_assembled_board_model',
                             'correct_assembled_physical_net',
                             'implemented_signal'))) = 1)
      OR
      (SIZEOF(USEDIN(SELF,
              specified_role('pcb_assembled_board_model',
                             'shorting_assembled_physical_net',
                             'shorted_signals'))) >= 1)
      OR
      (SIZEOF(USEDIN(SELF,
              specified_role('pcb_assembled_board_model',
                             'incorrect_non_shorting_assembled_physical_net',

component_terminal

  ENTITY component_terminal
      ABSTRACT SUPERTYPE OF (ONEOF(connector_component_terminal,
                                   electrical_component_terminal, 
                                   jumper_component_terminal));
    referenced_component : component; -- reference
    referenced_terminal  : part_terminal; -- reference
  UNIQUE
      uniqueness_constraint :
      (* There is only one component_terminal for a particular part_terminal on
         a component. *) 
      referenced_component, referenced_terminal;
  WHERE    
    valid_component_signal :
      (* A component_terminal is referenced by at most one component_signal and
         that component_signal is defined in the same internal_pcb_mcm_view
         as the component. *) 
      SIZEOF(QUERY(sig 
                   <* referenced_component.containing_view.component_signals |
             SELF IN sig.joined_terminals)) <= 1;

    valid_assembled_physical_net :
      (* A component_terminal is referenced by at most one
         assembled_physical_net and that assembled_physical_net is defined in 
         the assembled_board which implements the internal_pcb_mcm_view that
         owns the component. *)
      SIZEOF(QUERY(net <* 
        referenced_component.containing_view.is_implemented_by.physical_nets |
        SELF IN net.joined_terminals)) <= 1;
  END_ENTITY;

containing_signal_of_component_terminal_group

  FUNCTION containing_signal_of_component_terminal_group
    (terminalGroup : component_terminal_group) : component_signal;
    LOCAL
      componentSignals : BAG OF component_signal := [ ];
      orderedGroups    : BAG OF component_terminal_ordered_group := [ ];
      unorderedGroups  : BAG OF component_terminal_unordered_group := [ ];
    END_LOCAL;

    componentSignals := USEDIN(terminalGroup,
      'pcb_mcm_view_model.component_signal.terminal_groupings');

    IF (SIZEOF(componentSignals) = 1) THEN
      RETURN(componentSignals[1]);
    ELSE
      orderedGroups := USEDIN(terminalGroup,
      'pcb_mcm_view_model.component_terminal_ordered_group.sub_groupings');
      IF (SIZEOF(orderedGroups) = 1) THEN
          RETURN(containing_signal_of_component_terminal_group
                 (orderedGroups[1]));
      ELSE
        unorderedGroups := USEDIN(terminalGroup,
        'pcb_mcm_view_model.component_terminal_unordered_group.sub_groupings');
        IF (SIZEOF(unorderedGroups) = 1) THEN
            RETURN(containing_signal_of_component_terminal_group
                   (unorderedGroups[1]));
        ELSE
            RETURN(?);
        END_IF;
      END_IF;
    END_IF;
  END_FUNCTION;

      (* An allocated_component_function_terminal belongs to one
         allocated_component_function. *) 
      belongs_to_one(SELF,
        'pcb_mcm_view_model.allocated_component_function.allocated_terminals');

    valid_allocated_port :
      (* The allocated instance_port must refer to the instance which is
         allocated in the containing allocated_component_function. *)
      allocated_port.referenced_instance
        :=: containing_allocation.allocated_instance;

    valid_referenced_terminals :
      (* The referenced part_function_terminal must refer to the part_function
         which is referenced in the containing allocated_component_function.*) 
      not_exists(QUERY(functionTerminal <* referenced_terminals |
        functionTerminal.referenced_part_function
        :<>: containing_allocation\component_function_allocation
                                  .referenced_part_function));
  END_ENTITY;

Description:

An allocated_component_function_terminal describes how a terminal on a component function is allocated to a port on an instance.

Description:

A swap_indicator specifies the status of an allocation. If the status is fixed, it implies the allocation cannot be removed.
