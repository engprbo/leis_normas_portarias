assembled_board

  ENTITY assembled_board;
    is_based_on                        : bare_board;
    component_implementations          : OPTIONAL SET [1:?] OF
                                         component_implementation; 
    embedded_component_implementations : OPTIONAL SET [1:?] OF -- chips-first
                                         embedded_component_implementation;
    drawings                           : OPTIONAL SET [1:?] OF pcb_drawing;
    drawing_board_subsets              : OPTIONAL SET [1:?] OF board_subset;
    physical_nets                      : OPTIONAL SET [1:?] OF
                                         assembled_physical_net; 
    component_assembly_rules           : OPTIONAL SET [1:?] OF
                                         component_assembly_rule;
    component_orientation_rules        : OPTIONAL SET [1:?] OF -- reference
                                         mountable_package_orientation_rule; 
    component_realisation_rules        : OPTIONAL SET [1:?] OF
                                         component_realisation_rule;
    component_signal_realisation_rules : OPTIONAL SET [1:?] OF
                                         component_signal_realisation_rule;
    default_maximum_component_height_top_side    : OPTIONAL length_value;
    default_maximum_component_height_bottom_side : OPTIONAL length_value;
    wirebond_length                              : OPTIONAL toleranced_distance;
  DERIVE
    containing_view : internal_pcb_mcm_view -- reference
      := usedin_one_role(SELF,
         'pcb_mcm_view_model.internal_pcb_mcm_view.is_implemented_by');
  WHERE
    valid_existence :
      (* An assembled_board belongs to one internal_pcb_mcm_view. *)
      belongs_to_one(SELF,
        'pcb_mcm_view_model.internal_pcb_mcm_view.is_implemented_by');

    valid_nets :
      (* The assembled_physical_nets are either all analysed or all
         non-analysed. *)
      not_exists(QUERY(physicalNet1 <* physical_nets |
        there_exists(QUERY(physicalNet2 <* physical_nets - physicalNet1 |
          TYPEOF(physicalNet1) <> TYPEOF(physicalNet2)))));

    valid_implementation : -- chips-first
      (* A component in the containing view can be implemented at most once.
         That is, it cannot be implemented by a component_implementation and
         by an embedded_component_implementation at the same time. *)  
     not_exists(QUERY(component <* containing_view.components |
       (SIZEOF(QUERY(componentImpl <* component_implementations |
                     componentImpl.implemented_component :=: component))
        +
        SIZEOF(QUERY(componentImpl <* embedded_component_implementations |
                     componentImpl.implemented_component :=: component))
       ) > 1));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. An assembled_board cannot have component_implementations,
            embedded_component_implementations, component_assembly_rules or
            component_realisation_rules if its containing view does not define
            any components. 
         2. An assembled_board cannot have component_signal_realisation_rules
            if its containing view does not define any component_signals. *) 
      (EXISTS(containing_view.components) OR
         NOT EXISTS(component_implementations) AND
         NOT EXISTS(embedded_component_implementations) AND
         NOT EXISTS(component_assembly_rules) AND
         NOT EXISTS(component_realisation_rules)) 
      AND
      (EXISTS(containing_view.component_signals) OR
         NOT EXISTS(component_signal_realisation_rules));
  END_ENTITY;

bare_board

  ENTITY bare_board;
    instantiated_bare_board : bare_board_definition; -- reference
    property_overrides      : OPTIONAL SET [1:?] OF property_override;
  WHERE
    valid_existence :
      (* A bare_board belongs to one assembled_board. *)
      belongs_to_one(SELF,
        'pcb_assembled_board_model.assembled_board.is_based_on');

    valid_property_overrides :
      (* The overridden properties are defined in the instantiated
         bare_board_definition. *)
      not_exists(QUERY(propertyOverride <* property_overrides |
        NOT (propertyOverride.overridden_property
             IN instantiated_bare_board.properties)));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. A bare_board cannot have property_overrides if its instantiated
            bare_board_definition does not define any properties. *)
      EXISTS(instantiated_bare_board.properties) OR
        NOT EXISTS(property_overrides);
  END_ENTITY;

Description:

A bare_board is a single occurrence of a bare_board_definition object. A bare board is a composition of laminated sheets of conductive material, separated by insulation sheets, and usually interconnected at several places. The bare board is used in the assembly process, to mount packaged (pre-fabricated) components onto, and appears in the bill of material of the assembled board.

Used by:

assembled_board bare_board_instance_net

embedded_component_implementation

  ENTITY embedded_component_implementation;     implemented_component : component; -- reference
    implementing_die      : embedded_die; -- reference
  DERIVE
    containing_board : assembled_board -- reference
      := usedin_one_role(SELF,
                         specified_role('pcb_assembled_board_model',
                                        'assembled_board',
                                        'embedded_component_implementations'));
  UNIQUE
    single_implementation :
      (* An embedded_die in an assembled_board cannot be used to implement more
         than one component. *)
      containing_board, implementing_die;
  WHERE
    valid_existence :
      (* An embedded_component_implementation belongs to one
         assembled_board. *) 
      belongs_to_one(SELF,
                     specified_role('pcb_assembled_board_model',
                                    'assembled_board',
                                    'embedded_component_implementations'));

    valid_implemented_component :
      (* The implemented component must be defined in the containing pcb or
         mcm view. *)
      implemented_component IN containing_board.containing_view.components;

    valid_implementing_die :
      (* The referenced embedded_die must be defined in the
         bare_board_definition from which the containing assembled_board is
         instantiated. *)
      EXISTS(containing_board.is_based_on.instantiated_bare_board.embedded_dies)
      AND
      (implementing_die 
         IN containing_board.is_based_on.instantiated_bare_board.embedded_dies);

    valid_implementation :
      (* The implemented component must instantiate the same
         die_electrical_part as the embedded_die. *)
      implemented_component.instantiated_part
        :=: implementing_die.embedded_part;
  END_ENTITY;

component_implementation

  ENTITY component_implementation
      ABSTRACT SUPERTYPE OF (ONEOF (die_component_implementation,
                                    packaged_component_implementation,
                                    printed_component_implementation));
    implemented_component                     : component; -- reference
    referenced_footprint                      : footprint_instance; -- reference
    component_designator_fabrication_override : OPTIONAL 
                                             fabricate_text_appearance_override;
    component_property_fabrications           : OPTIONAL SET [1:?] OF
                                                property_fabrication;   
    part_property_fabrications                : OPTIONAL SET [1:?] OF
                                                property_fabrication;   
    drawing_information                       : OPTIONAL SET [1:?] OF
                                                drawing_element; 
  DERIVE
    containing_board : assembled_board -- reference
      := usedin_one_role(SELF,
         'pcb_assembled_board_model.assembled_board.component_implementations');
  WHERE
    valid_existence :
      (* A component_implementation belongs to one assembled_board. *)
      belongs_to_one(SELF,
        'pcb_assembled_board_model.assembled_board.component_implementations');

    valid_implemented_component :
      (* The implemented component must be defined in the containing pcb or mcm
         view. *)
      implemented_component IN containing_board.containing_view.components;

    valid_referenced_footprint :
      (* The referenced footprint_instance must be within the
         bare_board_definition which is instanced as the basis for the
         assembled_board. *)
      EXISTS(containing_board.is_based_on.instantiated_bare_board.sub_layouts)
      AND
      (referenced_footprint 
         IN containing_board.is_based_on.instantiated_bare_board.sub_layouts);

    valid_component_property_fabrications :
      (* The fabricated properties are defined in the implemented component. *)
      not_exists(QUERY(propertyFabrication <* component_property_fabrications |
        NOT (propertyFabrication.fabricated_property
             IN implemented_component.properties)));

    valid_part_property_fabrications :
      (* The fabricated properties are defined in the part referenced by the
         implemented component. *)
      not_exists(QUERY(propertyFabrication <* part_property_fabrications |
        NOT (propertyFabrication.fabricated_property
             IN implemented_component.instantiated_part.properties)));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. A component_implementation cannot have
            component_property_fabrication if its implemented component does
            not define any properties. 
         2. A component_implementation cannot have part_property_fabrications
            if the part from which the implemented component is instantiated
            does not define any properties. *)

      (EXISTS(implemented_component.properties) OR 
         NOT EXISTS(component_property_fabrications))
      AND
      (EXISTS(implemented_component.instantiated_part.properties) OR
         NOT EXISTS(part_property_fabrications));
  END_ENTITY;

         the part. *)

    'mcm_mountable_die_physical_model.internal_mountable_die_physical_template')

    (implemented_component.instantiated_part.referenced_die_physical_template
     :<>: mountable_template\internal_mountable_die_physical_template
          .referenced_die_physical_template));

    valid_component_terminal_implementations :
      (* The part_terminal referenced by the component_terminal_implementation
         must be defined in the part in which the component is instantiated. *)
      not_exists(QUERY(terminalImpl <* component_terminal_implementations |
        not_exists(QUERY(partTerminal <*
                         implemented_component.instantiated_part.terminals |
        terminalImpl\component_terminal_implementation.referenced_part_terminal 
        :=: partTerminal))));

    valid_overriding_mountable_template_properties :
      (* The overridden properties are defined in the mountable_template *)
      not_exists(QUERY(propertyOverride 
                       <* overriding_mountable_template_properties |
        NOT (propertyOverride.overridden_property
             IN mountable_template.properties)));

    valid_mountable_template_property_fabrications :
      (* The fabricated properties are defined in the
         mountable_die_physical_template used by the
         die_component_implementation. *) 
      not_exists(QUERY(propertyFabrication 
                       <* mountable_template_property_fabrications |
        NOT (propertyFabrication.fabricated_property
             IN mountable_template.properties)));

    valid_template_property_fabrications :
      (* The fabricated properties are defined in the die_physical_template
         referenced by the mountable_die_physical_template. *)
    NOT (
    is_typeof(mountable_template,
    'mcm_mountable_die_physical_model.internal_mountable_die_physical_template')

      NOT (propertyFabrication.fabricated_property
           IN mountable_template\internal_mountable_die_physical_template
              .referenced_die_physical_template.properties))));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. A die_component_implementation cannot have
            overriding_mountable_template_properties or
            mountable_template_property_fabrications if the referenced
            mountable_die_physical_template does not define any properties. *)
      EXISTS(mountable_template.properties) OR
        NOT EXISTS(overriding_mountable_template_properties) AND
        NOT EXISTS(mountable_template_property_fabrications);
  END_ENTITY;

  ENTITY shorting_assembled_physical_net
      SUBTYPE OF (incorrect_assembled_physical_net);
    shorted_signals : SET [1:?] OF component_signal; -- reference
  WHERE

         shorted have at least one of their component_terminals in the set of
         terminals actually joined by the assembled_physical_net. *) 
      not_exists(QUERY(shortedSignal <* shorted_signals |
        not_exists(QUERY(joinedTerminal <* shortedSignal.joined_terminals |
          joinedTerminal 
          IN SELF\assembled_physical_net.joined_terminals))));

         is not shorted by the assembled_physical_net. *)
      not_exists(QUERY(terminal <* SELF\assembled_physical_net.joined_terminals|
        there_exists(QUERY(sig <* SELF\assembled_physical_net.containing_board
                                      .containing_view.component_signals |

           AND
           NOT (sig IN shorted_signals)))));

      (* The dependencies of these optional attributes are stated as follows:
         1. A shorting_assembled_physical_net cannot have joined_terminals if
            none of the shorted signals joins any component_terminals. *)
      there_exists(QUERY(sig <* shorted_signals | EXISTS(sig.joined_terminals)))
        OR NOT EXISTS(SELF\assembled_physical_net.joined_terminals);

  ENTITY incorrect_non_shorting_assembled_physical_net
      SUBTYPE OF (incorrect_assembled_physical_net);
    should_implement : OPTIONAL component_signal; -- reference
  DERIVE
    signal_terminals  : SET OF component_terminal :=
                        should_implement.joined_terminals;
    desired_terminals : SET OF component_terminal :=
                        NVL (signal_terminals, []);
    extra_terminals   : SET OF component_terminal :=
                        joined_terminals - desired_terminals;
    missing_terminals : SET OF component_terminal :=
                        desired_terminals - joined_terminals;
  WHERE
    valid_incompletion_or_overkill :
      (* An incorrect, but non-shorting, assembled physical net either
         joins extra terminals which are not in the set specified by
         the component_signal, or does not join all of the terminals
         specified by the component_signal, or both at the same
         time. *)
      SIZEOF (extra_terminals) + SIZEOF (missing_terminals) > 0;

    valid_non_shorting :
      (* An incorrect non-shorting assembled_physical_net may join
         extra terminals which are not specified in the
         component_signal.  However, as it does not short
         component_signals, the extra terminals must not belong to a
         component_signal. *)
      not_exists(QUERY(extraTerminal <* extra_terminals |
        belongs_to_one(extraTerminal,
          'pcb_mcm_view_model.component_signal.joined_terminals')));
  END_ENTITY;

    instantiated_wirebond_template       : wirebond_template; -- reference
  WHERE
    valid_component_terminals :
      (* A jumper_wirebond may be used to connect terminals of
         die_electrical_components, or between padstack_instances. *)
      (SIZEOF(QUERY(electricalComponentTerminal
                    <* QUERY(jumperPoint <* jumper_points |
                       is_typeof(jumperPoint,
                       'pcb_mcm_view_model.electrical_component_terminal')) |
                   is_typeof(electricalComponentTerminal.referenced_component,
                             'pcb_mcm_view_model.die_electrical_component')))
       = SIZEOF(jumper_points))
      OR
      not_exists(QUERY(electricalComponentTerminal
                       <* QUERY(jumperPoint <* jumper_points |
                          is_typeof(jumperPoint,
                          'pcb_mcm_view_model.electrical_component_terminal')) |
                 is_typeof(electricalComponentTerminal.referenced_component,
                           'pcb_mcm_view_model.die_electrical_component')));
  END_ENTITY;

Description:

A jumper_wirebond is a wirebond which completes a connection in an assembled_board.

component_signal_realisation_rule

  ENTITY component_signal_realisation_rule;
    implemented_component_signals : SET [1:?] OF component_signal; -- reference
    layer_specific_rules          : OPTIONAL SET [1:?] OF 
                                    component_signal_layer_realisation_rule;
    minimum_spur_length           : OPTIONAL length_value;
    minimum_trace_width           : OPTIONAL length_value;
    optimum_trace_width           : OPTIONAL length_value;
    properties                    : OPTIONAL SET [1:?] OF property;
  DERIVE 
    containing_board : assembled_board -- reference
      := usedin_one_role(SELF,
           specified_role('pcb_assembled_board_model',
                          'assembled_board',
                          'component_signal_realisation_rules'));
  WHERE
    valid_existence :
      (* A component_signal_realisation_rule belongs to one assembled_board. *)
      belongs_to_one(SELF,
        specified_role('pcb_assembled_board_model',
                       'assembled_board',
                       'component_signal_realisation_rules'));

    valid_trace_width :
      (* The optimum width is greater than the minimum width. *)
      optimum_trace_width > minimum_trace_width;

    valid_implemented_component_signals :
      (* The referenced component_signals are defined in the containing
         pcb_mcm_view. *)
      implemented_component_signals 
        <= containing_board.containing_view.component_signals;

    valid_layer_specific_rules :
      (* Any layer referenced by a component_signal_layer_realisation_rule
         cannot be also referenced by another
         component_signal_layer_realisation_rule. *) 
      not_exists(QUERY(layerRule1 <* layer_specific_rules |
        there_exists(QUERY(layerRule2 <* layer_specific_rules - layerRule1 |
          there_exists(layerRule1.applied_layers 
                       * layerRule2.applied_layers)))));
  END_ENTITY;

Description:

A component_signal_realisation_rule is used to state the physical or clearance constraints for component_signals which are realized by assembled_physical_nets on an assembled_board. It may define the default minimum/optimum trace width. It is possible to override the default trace width on individual layer(s).

The minimum_spur_length is the smallest length of the trace that must be laid when routing from the main track of a signal to a load pin.
