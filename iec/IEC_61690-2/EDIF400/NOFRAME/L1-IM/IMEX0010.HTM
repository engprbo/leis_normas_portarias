frame_model

Description:

The frame_model schema is used to define frames. The usage of frames is to provide users the ability to repeatedly or conditionally include a circuit in a design. A frame may contain instances and nested frames. Its logical connectivity is defined by frame_signal and frame_signal_group.

frame

  ENTITY frame
      ABSTRACT SUPERTYPE OF (ONEOF(for_frame,
                                   if_frame,
                                   otherwise_frame));
    document            : OPTIONAL SET [1:?] OF documentation;
    frame_signals       : OPTIONAL SET [1:?] OF frame_signal;
    frame_signal_groups : OPTIONAL SET [1:?] OF frame_signal_group;
    instances           : OPTIONAL SET [1:?] OF instance;
    nested_frames       : OPTIONAL SET [1:?] OF frame;
    properties          : OPTIONAL SET [1:?] OF property;
  DERIVE
    containing_frame : frame -- reference
      := usedin_one_role(SELF,'frame_model.frame.nested_frames');
    containing_view  : cell_representation -- reference
      := containing_view_of_frame(SELF);
  WHERE
    containment_constraint :
      (* A frame may either belong to one internal_connectivity_view, one
         internal_schematic_view, or another frame. *)
      either_belongs_to_one(SELF,
       ['connectivity_view_model.internal_connectivity_view.frames',
        'schematic_view_model.internal_schematic_view.frames',
        'frame_model.frame.nested_frames']);

    valid_frame_signals :
      (* All frame_signals are disjoint.  That is only one frame_signal may
         connect to a particular global_port, instance_member_port_member,
         instance_port, master_port or master_port_member.  Also, the same port
         cannot be referenced by more than one frame_signal.  Therefore, if a
         wide port has been referenced by a frame_signal, it is illegal to
         reference members of that wide port by other frame_signals.
         Conversely, if a member port has been referenced by a frame_signal, it
         is illegal to reference the wide port containing that member port by
         another frame_signal. *)   

      not_exists(QUERY(sig1 <* frame_signals |
        there_exists(QUERY(sig2 <* frame_signals - sig1 |
          there_exists(sig1.joined_global_ports * sig2.joined_global_ports)
       OR there_exists(sig1.joined_instance_member_port_members 
                       * sig2.joined_instance_member_port_members)
       OR there_exists(sig1.joined_instance_ports * sig2.joined_instance_ports)
       OR there_exists(sig1.joined_master_ports * sig2.joined_master_ports)
       OR there_exists(sig1.joined_master_port_members 
                       * sig2.joined_master_port_members)))));

    valid_nested_frames :
      (* A frame cannot contain itself directly or indirectly. *)
      non_recursive_frame(SELF, nested_frames);

    valid_optional_attributes :
       (* The dependencies of these optional attributes are stated as follows:
          1. Signal groups cannot be defined if there are no signals defined. *)
       EXISTS(frame_signals) OR NOT EXISTS(frame_signal_groups);
  END_ENTITY;

otherwise_frame

  ENTITY otherwise_frame
      SUBTYPE OF (frame);
    related_frames : SET [1:?] OF if_frame; -- reference
  WHERE
    valid_related_frames :
      (* All related_frames are directly defined in the containing frame, or in
         the containing view if there is no containing frame. *)
      not_exists(QUERY(memberFrame <* related_frames |
        NOT ((USEDIN(SELF, 'frame_model.frame.nested_frames')
              = USEDIN(memberFrame, 'frame_model.frame.nested_frames'))
             OR
             (USEDIN(SELF, 
               'connectivity_view_model.internal_connectivity_view.frames')
              = USEDIN(memberFrame,
               'connectivity_view_model.internal_connectivity_view.frames'))
             OR
             (USEDIN(SELF, 
               'schematic_view_model.internal_schematic_view.frames')
              = USEDIN(memberFrame,
               'schematic_view_model.internal_schematic_view.frames')))));
  END_ENTITY;

Description:

 LEVEL 1
 =======
An otherwise_frame is a frame. The contents of an otherwise_frame are conditionally included in the expansion if the conditions of all related if_frames evaluate to FALSE.

Used by:

schematic_otherwise_frame_implementation

containing_view_of_frame

  FUNCTION containing_view_of_frame
    (nestedFrame : frame) : cell_representation;
    IF EXISTS(nestedFrame.containing_frame) THEN
       RETURN(nestedFrame.containing_frame.containing_view);
    ELSE
       RETURN(usedin_any_one_role(nestedFrame,
              ['connectivity_view_model.internal_connectivity_view.frames', 
               'schematic_view_model.internal_schematic_view.frames']));
    END_IF;
  END_FUNCTION;

frame_signal

  ENTITY frame_signal
      ABSTRACT SUPERTYPE OF (ONEOF(frame_mergeable_signal,
                                   frame_non_mergeable_signal));
    joined_global_ports                 : OPTIONAL SET [1:?] OF -- reference
                                          global_port;
    joined_instance_member_port_members : OPTIONAL SET [1:?] OF
                                          instance_member_port_member;
    joined_instance_ports               : OPTIONAL SET [1:?] OF -- reference
                                          instance_port; 
    joined_master_ports                 : OPTIONAL SET [1:?] OF -- reference
                                          master_port;
    joined_master_port_members          : OPTIONAL SET [1:?] OF
                                          master_port_member;
    instance_port_expressions           : OPTIONAL SET [1:?] OF 
                                          instance_port_expression; 
    name                                : OPTIONAL name_information;  
    properties                          : OPTIONAL SET [1:?] OF property;
    width                               : integer_expression;
  DERIVE
    containing_frame      : frame   -- reference
      := usedin_one_role(SELF,'frame_model.frame.frame_signals');
    containing_cluster    : cluster -- reference
      := containing_frame.containing_view.containing_cluster;
    containing_edif       : edif    -- reference
      := containing_cluster.containing_cell.containing_library.containing_edif;
  WHERE
    containment_constraint :
      (* A frame_signal belongs to one frame. *) 
      belongs_to_one(SELF, 'frame_model.frame.frame_signals');

    valid_joined_global_ports :
      (* All joined global_ports are defined in the containing edif of the
         frame_signal. *) 
      joined_global_ports <= containing_edif.global_ports;

    valid_joined_instance_member_port_members :
      (* All joined instance_member_port_members refer to the instances defined
         directly in the immediately containing frame.  *)
      not_exists(QUERY(port <* joined_instance_member_port_members |
        NOT (port.referenced_instance IN containing_frame.instances)));

    valid_joined_instance_ports :
      (* All joined instance_ports refer to the instances defined directly
         inside the immediately containing frame. *)  
      not_exists(QUERY(port <* joined_instance_ports |
        NOT (port.referenced_instance IN containing_frame.instances)));

    valid_joined_master_ports :
      (* All joined master_ports are defined in the interface of the cluster
         containing the frame_signal.  No joined master_port is unused
         internally. *) 
      (joined_master_ports <= containing_cluster.interface.ports)
        AND
      not_exists(QUERY(port <* joined_master_ports | port.unused_internally));

    valid_joined_master_port_members :
      (* All joined master_port_members refer to the master_ports defined in
         the interface of the cluster containing the frame_signal.  No joined
         master_port_member is unused internally. *)
      not_exists(QUERY(port <* joined_master_port_members |
        NOT (port.referenced_master_port 
             IN containing_cluster.interface.ports)));

    valid_joined_wide_instance_ports :
      (* The same port cannot be referenced more than once within a signal.
         Therefore if a wide instance_port has been referenced, it is illegal
         to reference members of that wide instance_port.  Conversely, if a
         member port has been referenced, it is illegal to reference the wide
         instance_port containing that member port. *)
      not_exists(QUERY(joinedWidePort <* joined_instance_ports |
        there_exists(QUERY(joinedMemberPort 
                           <* joined_instance_member_port_members |
          joinedWidePort :=: joinedMemberPort.referenced_instance_port))));

    valid_joined_wide_master_ports :
      (* The same port cannot be referenced more than once within a signal.
         Therefore if a wide master_port has been referenced, it is illegal to
         reference members of that wide master_port.  Conversely, if a member
         port has been referenced, it is illegal to reference the wide
         master_port containing that member port. *)
      not_exists(QUERY(joinedWidePort <* joined_master_ports |
        there_exists(QUERY(joinedMemberPort <* joined_master_port_members |
          joinedWidePort :=: joinedMemberPort.referenced_master_port))));

    valid_width :
      (* The width expression may reference the for_frame_indexes of any
         containing for_frames or the parameters defined in the interface of
         the containing_view. *)  
      valid_integer_expression([width],
        containing_cluster.interface.parameters,
        find_nested_for_frame_index(containing_frame),
        ['expression_model.frame_signal_index',
         'expression_model.instance_index',
         'expression_model.port_index',
         'expression_model.signal_index']);

    valid_optional_attributes :
       (* The dependencies of these optional attributes are stated as follows:
          1. A signal cannot join any global_ports if there are no global_ports
             defined in the containing edif.  
          2. A signal cannot join any instance_ports or
             instance_member_port_members if there are no instances defined in
             the containing frame. 
          3. A signal cannot join any master_ports or master_port_members if
             there are no master_ports defined in the containing cluster. *)
       (EXISTS(containing_edif.global_ports) OR NOT EXISTS(joined_global_ports))
        AND
       (EXISTS(containing_frame.instances) OR
          NOT EXISTS(joined_instance_ports) AND
          NOT EXISTS(joined_instance_member_port_members))
        AND
       (EXISTS(containing_cluster.interface.ports) OR 
          NOT EXISTS(joined_master_ports) AND
          NOT EXISTS(joined_master_port_members));
  END_ENTITY;

Description:

 LEVEL 1
 =======
A frame_signal describes a signal defined within a frame. It is defined as a named object which lists all the global_ports, instance_ports, instance_member_port_members, master_ports and master_port_members which are electrically common at a given level of hierarchy. However, it can join instance_ports on instances directly inside the containing frame only.

frame_mergeable_signal

  ENTITY frame_mergeable_signal
      SUBTYPE OF (frame_signal);
    connectivity_tag_generator : string_expression;
  UNIQUE
    unique_connectivity_tag_generator :
      (* All frame_mergeable_signals defined in the same containing_frame must
         have unique connectivity_tag_generator. *)
      connectivity_tag_generator, SELF\frame_signal.containing_frame;
  WHERE
    valid_connectivity_tag_generator :
      (* The connectivity_tag_generator expression may reference the
         frame_signal_index of the frame_signal, the for_frame_indexes of any
         containing for_frames, or the parameters defined in the
         cluster_interface of the containing view. *) 
      valid_string_expression([connectivity_tag_generator],
                               containing_cluster.interface.parameters,
                               find_nested_for_frame_index(containing_frame),
                               ['expression_model.instance_index',
                                'expression_model.port_index',
                                'expression_model.signal_index']);
  END_ENTITY;

Description:

A frame_mergeable_signal is a frame_signal which can be merged with other mergeable_signals or frame_mergeable_signals. It has a connectivity_tag_generator which is vital for the merging process in the expansion of a design. After the expansion, the connectivity_tag_generator expression will be evaluated to a tag and all mergeable_signals or frame_mergeable_signals with the same tag are merged together. Within a frame, all frame_mergeable_signals must have unique connectivity_tag_generator. The connectivity_tag_generator expression may reference to the index of the frame_mergeable_signal, the for_frame_indexes of any containing for_frame or any parameter defined in the cluster_interface of the containing view.

frame_signal_group

  ENTITY frame_signal_group;
    members    : LIST [1:?] OF frame_signal_or_frame_signal_group;  -- reference
    name       : OPTIONAL name_information;  
    properties : OPTIONAL SET [1:?] OF private_property;
  DERIVE
    containing_frame : frame -- reference
      := usedin_one_role(SELF,'frame_model.frame.frame_signal_groups');
    flattened_signal_list : LIST OF frame_signal -- reference
                            := flatten_frame_signal_group(members);
  UNIQUE
    unique_members :
      (* No two frame_signal_groups have the same structure of members. *)
      members;
  WHERE
    containment_constraint :
      (* A frame_signal_group belongs to one frame. *) 
      belongs_to_one(SELF, 'frame_model.frame.frame_signal_groups');

    valid_members :
      (* All member frame_signals or frame_signal_groups are defined in the
         immediately containing frame. *)
      not_exists(QUERY(member <* members |
        NOT ((member IN containing_frame.frame_signals) 
             OR (member IN containing_frame.frame_signal_groups))));

    non_recursive_constraint :
      (* A frame_signal_group cannot contain itself directly or indirectly. *)
      not_recursive_frame_signal_group([SELF], [SELF]);
  END_ENTITY;

flatten_frame_signal_group

  FUNCTION flatten_frame_signal_group
    (signal_group_members : LIST OF frame_signal_or_frame_signal_group) 
    : LIST OF frame_signal;
    LOCAL
      position    : INTEGER := 1;
      signal_list : LIST OF frame_signal := [ ];
      member_list : LIST OF frame_signal := [ ];
    END_LOCAL;

    REPEAT i := 1 TO HIINDEX(signal_group_members);
     IF 'frame_model.frame_signal' IN TYPEOF(signal_group_members[i]) THEN
       INSERT(signal_list, signal_group_members[i], position);
       position := position + 1;
     ELSE
       member_list := flatten_frame_signal_group
                      (signal_group_members[i].members);
       REPEAT j := 1 TO SIZEOF(member_list);
         INSERT(signal_list, member_list[j], position);
         position := position + 1;
       END_REPEAT;
     END_IF;
    END_REPEAT;
    RETURN(signal_list);
  END_FUNCTION;

not_recursive_frame_signal_group

  FUNCTION not_recursive_frame_signal_group 
    (parents, invalid_ancestors : SET OF frame_signal_group) : BOOLEAN;
    LOCAL
      i, j        : INTEGER;
      descendants : SET OF frame_signal_group := [ ];
    END_LOCAL;
    REPEAT i := 1 TO HIINDEX(parents);
      REPEAT j := 1 TO HIINDEX(parents[i].members);
        IF 'frame_model.frame_signal_group' 
           IN TYPEOF(parents[i].members[j]) THEN
         descendants := descendants + parents[i].members[j];
        END_IF;
      END_REPEAT;
    END_REPEAT;
    RETURN
     (not_exists(descendants) OR
      (not_exists(descendants * invalid_ancestors) AND
       not_recursive_frame_signal_group(descendants, 
                                        parents+invalid_ancestors)));
  END_FUNCTION;

Description:

 LEVEL 1
 =======
The not_recursive_frame_signal_group function returns true if a frame_signal_group does not contain members which reference itself. It prevents cyclic definition of a frame_signal_group.

Used by:

frame_signal_group

frame_signal_group_member

  FUNCTION frame_signal_group_member
    (member, parent : frame_signal_group) : BOOLEAN;
    LOCAL
      is_member : BOOLEAN := FALSE;
      i         : INTEGER;
    END_LOCAL;
    IF member IN parent.members THEN
       RETURN(TRUE);
    ELSE
       i := 1;
       REPEAT WHILE (is_member = FALSE) AND (i <= SIZEOF(parent.members));
         IF ('frame_model.frame_signal_group' 
             IN TYPEOF(parent.members[i])) THEN
            is_member := frame_signal_group_member(member, parent.members[i]);
         ELSE
            i := i + 1;
         END_IF;
       END_REPEAT;
       RETURN(is_member);
    END_IF;
  END_FUNCTION;

Description:

The frame_signal_group_member function returns true if a frame_signal_group is a member of another frame_signal_group, either directly or indirectly.

Used by:

connectivity_frame_bus_slice schematic_frame_bus_slice

non_recursive_frame

  FUNCTION non_recursive_frame
    (parentFrame  : frame;
     nestedFrames : SET OF frame) : BOOLEAN;
     REPEAT i := 1 TO SIZEOF(nestedFrames);
       IF parentFrame :=: nestedFrames[i] THEN
          RETURN(FALSE);
       ELSE
       IF 'frame_model.otherwise_frame' IN TYPEOF(nestedFrames[i]) THEN
          IF NOT non_recursive_frame(parentFrame,
                 nestedFrames[i]\otherwise_frame.related_frames) THEN
             RETURN(FALSE);
          END_IF;
       END_IF;
       END_IF;
     END_REPEAT;
     RETURN(TRUE);     
  END_FUNCTION;

Description:

The non_recursive_frame returns true if a frame does not contain itself, either directly or indirectly.

Used by:

frame

find_nested_for_frame_index

  FUNCTION find_nested_for_frame_index
    (fr : frame) : SET OF for_frame_index;
    LOCAL
      indexes : SET OF for_frame_index := [ ];
    END_LOCAL;
    IF 'frame_model.for_frame' IN TYPEOF(fr) THEN
       indexes := indexes + fr\for_frame.index;
    END_IF;
    IF EXISTS(fr.containing_frame) THEN
       indexes := indexes + find_nested_for_frame_index(fr.containing_frame);
    END_IF;
    RETURN(indexes);
  END_FUNCTION;

Description:

The find_nested_for_frame_index function returns all for_frame_index defined in the current for_frame and also in any for_frames containing the current for_frame either directly or indirectly.
