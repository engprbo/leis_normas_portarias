closed_area_element_with_cutout

  ENTITY closed_area_element_with_cutout
      SUBTYPE OF (closed_area_element);
    outline : closed_area_element;
    cutouts : SET [1:?] OF closed_area_element; -- cutout boundaries
  WHERE
    valid_outline :
      (* The outline cannot be a closed_area_element_with_cutout. *)
      NOT('geometry_model.closed_area_element_with_cutout'
          IN TYPEOF(outline));

    valid_cutouts :
      (* A cutout may intersect its outline at a single point but may not
         intersect at more than one point or go outside the outline.  No cutout
         can overlap or intersect any other cutout of the same
         closed_area_element_with_cutout. *) 
      not_exists(QUERY(cutout <* cutouts |
        NOT closed_area_element_inside_closed_area_element(cutout, outline)))
      AND
      not_exists(QUERY(c1 <* cutouts |
        there_exists(QUERY(c2 <* cutouts - c1 |
          closed_area_element_inside_closed_area_element(c1, c2)
          OR
          closed_area_element_inside_closed_area_element(c2, c1)
          OR
          closed_area_element_intersect_closed_area_element(c1, c2)))));
  END_ENTITY;

Description:

A closed_area_element_with_cutout describes a geometry_element which encloses an area. It is defined as the area enclosed by a given outline boundary, less the areas enclosed by a given set of one or more cutout boundaries. Each outline and cutout boundary may be a closed_area_element.

polygon

  ENTITY polygon
      SUBTYPE OF (closed_area_element);
    segments : LIST [3:?] OF line_segment;
  DERIVE
    vertices : LIST OF integer_point
               := find_polygon_point(segments);
  WHERE                                                                       
    joined_segments :
      (* All adjacent segments are joined. *)
      connected(segments);

    closed_element : 
      (* The first segment is joined with the last segment. *)
      segments[1].start_point = segments[HIINDEX(segments)].end_point;

      (* A polygon may not intersect or touch itself, nor have coincidence
         segments. *) 
      NOT intersecting_or_touching_or_coincident(segments);
  END_ENTITY;

arc

  ENTITY arc ;
    start_point   : integer_point;
    through_point : number_point;
    end_point     : integer_point;
  DERIVE
    centre_point  : point := centre_point_of_arc(SELF);
    radius        : distance_value := point_distance(centre_point,start_point);
  WHERE
    distinct_points :
      (* The three points of the arc specification must be distinct. *)
      ((start_point.x_coordinate <> through_point.x_coordinate)
        OR (start_point.y_coordinate <> through_point.y_coordinate))
      AND      
      ((start_point.x_coordinate <> end_point.x_coordinate)
        OR (start_point.y_coordinate <> end_point.y_coordinate))
      AND 
      ((through_point.x_coordinate <> end_point.x_coordinate)
        OR (through_point.y_coordinate <> end_point.y_coordinate));

    valid_points :
      (* The three points should not lie on a straight line. *)
      NOT colinear_points(start_point, end_point, through_point);
  END_ENTITY;

Description:

An arc is used in describing shapes with curved edges. It is defined by three points. Each point coordinate is specified in distance value. The point coordinate values of the starting point and the end point of the arc must be integers. The point coordinate values of the middle point of the arc can be given as numbers.

  ENTITY arc_by_centre;
    centre_point        : integer_point;
    clockwise_direction : boolean_token;
    end_point           : integer_point;
    start_point         : integer_point;

connected

  FUNCTION connected 
    (segments : LIST OF arc_or_line_segment) : BOOLEAN;
    LOCAL
      i       : INTEGER;
    END_LOCAL;

    REPEAT i := 1 TO HIINDEX(segments)-1;
      IF segments[i].end_point <> segments[i+1].start_point THEN
         RETURN(FALSE);
      END_IF;
    END_REPEAT;
    RETURN(TRUE);  
  END_FUNCTION;

Description:

The connected function accepts a list of arc or line_segment elements as its argument. It checks that the last point of each arc or line_segment element is the same as the first point of the next arc or line_segment element on the list.

Used by:

colinear_points

  FUNCTION colinear_points
    (p1, p2, p3: point) : BOOLEAN;
    LOCAL
      m, c : NUMBER;
    END_LOCAL;

     m := (p2.y_coordinate - p1.y_coordinate) /
          (p2.x_coordinate - p1.x_coordinate);
     c :=  p1.y_coordinate - m*p1.x_coordinate;

     RETURN(p3.y_coordinate = (m*p3.x_coordinate + c));
  END_FUNCTION;

Description:

The colinear_points function returns true if three points lie on a straight line.

The function calculates the equation of the line for the start and end points (i.e. p1, p2). Then, it checks whether or not the third point (i.e. p3) satisfies the equation. If it does, then p1, p2 and p3 are colinear. The formula of a line is represented as below: 

 y = m*x + c
 where m = (p2.y - p1.y)/(p2.x - p1.x) and c = p1.y - m*p1.x

Used by:

arc

point_and_line_distance

  FUNCTION point_and_line_distance
    (C : point; l : line_segment) : distance_value;

    -- The algorithm is based on the FAQ of the comp.graphics.algorithms
    -- newsgroup                                                       
    --                                                                 
    --    Let the point be C (XC,YC) and the line be AB (XA,YA) to
    --    (XB,YB). The length of the line segment AB is L:
    --                                                                 
    --                (YA-YC)(YA-YB)-(XA-XC)(XB-XA)
    --            r = -----------------------------
    --                            L**2
    --                                                                 
    --    Let I be the point of perpendicular projection of C onto AB
    --            XI=XA+r(XB-XA)
    --            YI=YA+r(YB-YA)

    LOCAL
      A, B : integer_point;
      I    : number_point;
      r    : NUMBER;
    END_LOCAL;

    A := l.start_point;
    B := l.end_point;

    r := ((A.y_coordinate-C.y_coordinate)*(A.y_coordinate-B.y_coordinate) -
          (A.x_coordinate-C.x_coordinate)*(B.x_coordinate-A.x_coordinate))
         /
         ((B.x_coordinate-A.x_coordinate)**2 + 
          (B.y_coordinate-A.y_coordinate)**2);
  
    I.x_coordinate := A.x_coordinate + r * (B.x_coordinate - A.x_coordinate);
    I.y_coordinate := A.y_coordinate + r * (B.y_coordinate - A.y_coordinate);

    RETURN(point_distance(C, I));
  END_FUNCTION;

Description:

The point_and_line_distance returns the distance between a point to a line.

  FUNCTION intersecting_or_touching_or_coincident
    (segments : LIST OF arc_or_line_segment) : BOOLEAN;

    LOCAL
      n                   : INTEGER := SIZEOF(segments);
      adj_pairs           : SET OF SET [2:2] OF INTEGER := [ ];
      opp_pairs           : SET OF SET [2:2] OF INTEGER := [ ];
      s1, s2              : arc_or_line_segment;
      intersecting_points : SET OF point;
    END_LOCAL;

    -- Finding pairs of arc_or_line_segments which are adjacent to each other.
    -- For example, if there is 5 segments, adjacent pairs are [1,2], [2,3],
    -- [3,4], [4,5] and [1,5].

    REPEAT i := 1 TO n-1;
      adj_pairs := adj_pairs + [[i,i+1]];
    END_REPEAT;
    adj_pairs := adj_pairs + [[1,n]];

    -- Finding pairs of arc_or_line_segments which are not adjacent to each
    -- other.  For example, if there is 5 segments, non-adjacent pairs are
    -- [1,3], [1,4], [2,4], [2,5] and [3,5].

    REPEAT i := 1 TO n;
      REPEAT j := 1 TO n;
        opp_pairs := opp_pairs + [[i,j]];
      END_REPEAT;
      opp_pairs := opp_pairs - [[i,i]];
      CASE i OF
        1         : opp_pairs := opp_pairs - [[1,2], [1,n]];
        n         : opp_pairs := opp_pairs - [[1,n], [n-1,n]];
        OTHERWISE : opp_pairs := opp_pairs - [[i-1,i], [i,i+1]];
      END_CASE;
    END_REPEAT;

    -- Checking whether non-adjacent pairs of arc_or_line_segments are
    -- intersecting, touching or coincident.

    REPEAT i := 1 TO SIZEOF(opp_pairs);
      s1 := segments[opp_pairs[i][1]];
      s2 := segments[opp_pairs[i][2]];
      IF is_typeof(s1, 'geometry_model.line_segment') AND
         is_typeof(s2, 'geometry_model.line_segment') THEN
         IF line_touching_line(s1, s2) OR
            line_intersect_line(s1, s2) THEN
            RETURN(TRUE);
         END_IF;
      ELSE
      IF is_typeof(s1, 'geometry_model.arc') OR
         is_typeof(s1, 'geometry_model.arc_by_centre') AND
         is_typeof(s2, 'geometry_model.line_segment') THEN
         IF arc_touching_line(s1, s2) OR
            arc_intersect_line(s1, s2) THEN
            RETURN(TRUE);
         END_IF;
      ELSE
      IF is_typeof(s1, 'geometry_model.line_segment') AND
         is_typeof(s2, 'geometry_model.arc') OR
         is_typeof(s2, 'geometry_model.arc_by_centre') THEN
         IF arc_touching_line(s2, s1) OR
            arc_intersect_line(s2, s1) THEN
            RETURN(TRUE);
         END_IF;
      ELSE
      IF is_typeof(s1, 'geometry_model.arc') OR
         is_typeof(s1, 'geometry_model.arc_by_centre') AND
         is_typeof(s2, 'geometry_model.arc') OR
         is_typeof(s2, 'geometry_model.arc_by_centre') THEN
         IF arc_touching_arc(s1, s2) OR
            arc_intersect_arc(s1, s2) THEN
            RETURN(TRUE);
         END_IF;
      END_IF;
      END_IF;
      END_IF;
      END_IF;
    END_REPEAT;

    -- Checking whether adjacent pairs of arc_or_line_segments are intersecting,
    -- touching or coincident.  

    REPEAT i := 1 TO SIZEOF(adj_pairs);
      s1 := segments[adj_pairs[i][1]];
      s2 := segments[adj_pairs[i][2]];
      IF is_typeof(s1, 'geometry_model.line_segment') AND
         is_typeof(s2, 'geometry_model.line_segment') THEN
         IF point_on_line(s1.start_point, s2) OR
            point_on_line(s2.end_point, s1) THEN
            RETURN(TRUE);
         END_IF;
      ELSE
      IF is_typeof(s1, 'geometry_model.arc') OR
         is_typeof(s1, 'geometry_model.arc_by_centre') AND
         is_typeof(s2, 'geometry_model.line_segment') THEN
         IF arc_intersect_line(s1, s2) THEN
            RETURN(TRUE);
         END_IF;
      ELSE
      IF is_typeof(s1, 'geometry_model.line_segment') AND
         is_typeof(s2, 'geometry_model.arc') OR
         is_typeof(s2, 'geometry_model.arc_by_centre') THEN
         IF arc_intersect_line(s2, s1) THEN
            RETURN(TRUE);
         END_IF;
      ELSE
      IF is_typeof(s1, 'geometry_model.arc') OR
         is_typeof(s1, 'geometry_model.arc_by_centre') AND
         is_typeof(s2, 'geometry_model.arc') OR
         is_typeof(s2, 'geometry_model.arc_by_centre') THEN
         IF arc_intersect_arc(s1, s2) THEN
            RETURN(TRUE);
         END_IF;
      END_IF;
      END_IF;
      END_IF;
      END_IF;
    END_REPEAT;

    RETURN(FALSE);
  END_FUNCTION;

  FUNCTION arc_intersect_arc
    (a1, a2 : generic_arc) : BOOLEAN;
    LOCAL
      pts : SET OF point;
    END_LOCAL;

    -- Find the intersection points of two circles.  The circle is
    -- specified by the centre_point and the radius of the circular arc.
    pts := circle_and_circle_intersection(a1.centre_point, a1.radius, 
                                          a2.centre_point, a2.radius);

    -- The function returns true if there is any intersection point which lies
    -- on both arcs.  In addition, the point cannot be the start points or the
    -- end points of the two arcs. 
    IF SIZEOF(pts) = 2 THEN
       REPEAT i := 1 TO SIZEOF(pts);
         IF point_on_arc(pts[i], a1) AND point_on_arc(pts[i], a2) AND
            NOT (pts[i] IN [a1.start_point, a1.end_point, 
                            a2.start_point, a2.end_point]) THEN
            RETURN(TRUE);
         END_IF;
       END_REPEAT;
    END_IF;
    RETURN(FALSE);
  END_FUNCTION;

The arc_intersect_arc function returns true if the two arcs intersect.

  FUNCTION arc_intersect_line
    (a : generic_arc; l : line_segment) : BOOLEAN;
    LOCAL
      pts : SET OF point;
    END_LOCAL;

    -- Find the intersection points of a circle and a line.  The circle is
    -- specified by the centre_point and the radius of the circular arc.
    pts := circle_and_line_intersection(a.centre_point, a.radius, l);

    -- The function returns true if there is any intersection point which lies
    -- on both the line and the arc.  In addition, the point cannot be the
    -- start points or the end points of the line and the arc.
    IF SIZEOF(pts) = 2 THEN
       REPEAT i := 1 TO SIZEOF(pts);
         IF point_on_line(pts[i], l) AND point_on_arc(pts[i], a) AND
            NOT (pts[i] IN [a.start_point, a.end_point, 
                            l.start_point, l.end_point]) THEN
            RETURN(TRUE);
         END_IF;
       END_REPEAT;
    END_IF;
    RETURN(FALSE);
  END_FUNCTION;

line_intersect_line

  FUNCTION line_intersect_line
    (line1, line2 : line_segment) : BOOLEAN;

    -- The algorithm is based on the FAQ of the comp.graphics.algorithms
    -- newsgroup                                                       
    --                                                                 
    -- Let A,B,C,D be 2-space position vectors.  Then the directed line
    -- segments AB & CD are given by:                                  
    --                                                                 
    --    AB=A+r(B-A), r in [0,1]                                      
    --    CD=C+s(D-C), s in [0,1]                                      
    --                                                                 
    -- If AB & CD intersect, then                                      
    --                                                                 
    --    A+r(B-A)=C+s(D-C), or                                        
    --                                                                 
    --    XA+r(XB-XA)=XC+s(XD-XC)                                      
    --    YA+r(YB-YA)=YC+s(YD-YC)  for some r,s in [0,1]               
    --                                                                 
    -- Solving the above for r and s yields                            
    --                                                                 
    --        (YA-YC)(XD-XC)-(XA-XC)(YD-YC)                            
    --    r = -----------------------------                            
    --        (XB-XA)(YD-YC)-(YB-YA)(XD-XC)                            
    --                                                                 
    --        (YA-YC)(XB-XA)-(XA-XC)(YB-YA)                            
    --    s = -----------------------------                            
    --        (XB-XA)(YD-YC)-(YB-YA)(XD-XC)                            
    --                                                                 
    -- Let I be the position vector of the intersection point, then    
    --                                                                 
    --    I=A+r(B-A) or                                                
    --                                                                 
    --    XI=XA+r(XB-XA)                                               
    --    YI=YA+r(YB-YA)                                               
    --                                                                 
    -- If 0<r<1 & 0<s<1, intersection exists                       

    LOCAL
      r, s : NUMBER;
      x, y, denominator : INTEGER;
      A, B, C, D : integer_point;
    END_LOCAL;

    A := line1.start_point;
    B := line1.end_point;
    C := line2.start_point;
    D := line2.end_point;

    denominator := (B.x_coordinate - A.x_coordinate)
                    * (D.y_coordinate - C.y_coordinate)
                    - (B.y_coordinate - A.y_coordinate)
                       *(D.x_coordinate - C.x_coordinate);

    r := ((A.y_coordinate - C.y_coordinate)*(D.x_coordinate - C.x_coordinate) -
          (A.x_coordinate - C.x_coordinate)*(D.y_coordinate - C.y_coordinate))
         / denominator;
   
    s := ((A.y_coordinate - C.y_coordinate)*(B.x_coordinate - A.x_coordinate) -
          (A.x_coordinate - C.x_coordinate)*(B.y_coordinate - A.y_coordinate)) 
         / denominator;

    RETURN({0 < r < 1} AND {0 < s < 1});
  END_FUNCTION;

Description:

The line_intersect_line function returns true if the two lines intersect.

  FUNCTION arc_touching_arc
    (a1, a2 : generic_arc) : BOOLEAN;
    LOCAL
      pts : SET OF point;
    END_LOCAL;

    -- Find the intersection points of two circles.  The circle is
    -- specified by the centre_point and the radius of the circular arc.
    pts := circle_and_circle_intersection(a1.centre_point, a1.radius, 
                                          a2.centre_point, a2.radius);

    IF (SIZEOF(pts) = 1) AND point_on_arc(pts[1],a1) AND 
       point_on_arc(pts[1],a2) THEN
       RETURN(TRUE);
    ELSE
    IF SIZEOF(pts) = 2 THEN
       REPEAT i := 1 TO SIZEOF(pts);
         IF point_on_arc(pts[i], a1) AND point_on_arc(pts[i], a2) AND
            (pts[i] IN [a1.start_point, a1.end_point, 
                        a2.start_point, a2.end_point]) THEN
            RETURN(TRUE);
         END_IF;
       END_REPEAT;
    END_IF;
    END_IF;
    RETURN(FALSE);
  END_FUNCTION;

  FUNCTION point_inside_polygon
    (pt : point; p : generic_polygon) : BOOLEAN;

    -- The algorithm is based on the FAQ of the comp.graphics.algorithms
    -- newsgroup 

    LOCAL
      inside : BOOLEAN := FALSE;
      j, numOfVertices : INTEGER;
      v : LIST OF integer_point := p.vertices;
    END_LOCAL;

    numOfVertices := SIZEOF(v);
    j :=  numOfVertices - 1;

    REPEAT i := 0 TO (numOfVertices - 1);
      IF ({v[i].y_coordinate <= pt.y_coordinate < v[j].y_coordinate} OR
          {v[j].y_coordinate <= pt.y_coordinate < v[i].y_coordinate})
         AND
         ((v[j].y_coordinate - v[i].y_coordinate)
             / (v[j].x_coordinate - v[i].x_coordinate) 
          <
          (pt.y_coordinate - v[i].y_coordinate)
             / (pt.x_coordinate - v[i].x_coordinate)) THEN
         inside := NOT inside;
      END_IF;
      j := i;
    END_REPEAT;
    RETURN(inside);
  END_FUNCTION;

  FUNCTION angle_of_line
    (p1, p2 : point) : NUMBER;
    LOCAL
      xdiff : NUMBER := p2.x_coordinate - p1.x_coordinate;
      ydiff : NUMBER := p2.y_coordinate - p1.y_coordinate;
      angle : NUMBER;
    END_LOCAL;

    IF (xdiff = 0) AND (ydiff = 0) THEN
       RETURN(?);
    END_IF;

    IF (xdiff > 0) AND (ydiff = 0) THEN
       RETURN(0);
    END_IF;

    IF (xdiff = 0) AND (ydiff > 0) THEN
       RETURN(90);
    END_IF;

    IF (xdiff < 0) AND (ydiff = 0) THEN
       RETURN(180);
    END_IF;

    IF (xdiff = 0) AND (ydiff < 0) THEN
       RETURN(270);
    END_IF;

    angle := ATAN(ydiff, xdiff) * 180 / PI;
    IF (xdiff > 0) AND (ydiff > 0) THEN
       RETURN(angle);
    END_IF;

    IF (xdiff < 0) AND (ydiff > 0) THEN
       RETURN(180 + angle);
    END_IF;

    IF (xdiff < 0) AND (ydiff < 0) THEN
       RETURN(180 + angle);
    END_IF;

    IF (xdiff > 0) AND (ydiff < 0) THEN
       RETURN(360 + angle);
    END_IF;
  END_FUNCTION;

  FUNCTION centre_point_of_arc
    (a : arc) : point;

    -- Two lines can be formed through 2 pairs of the three points, the first
    -- passes through points p1 and p2 and the second passes through points p2
    -- and p3. 
    --
    -- The equation of these two lines are:
    -- L1: y = m1(x - P1.x) + P1.y where m1=(P2.y-P1.y)/(P2.x-P1.x)
    -- L2: y = m2(x - P2.x) + P2.y where m1=(P3.y-P2.y)/(P2.x-P2.x)
    --
    -- The centre of the circle is the intersection of the two lines
    -- perpendicular to and passing through the midpoints of lines L1 and L2.
    --
    -- The perpendicular of a line with slope m has slope -1/m, thus equations
    -- of lines perpendicular to lines L1 and L2 and passing through the
    -- midpoints of P1P2 and P2P3 are:
    -- LP1: y = -1/m1 *(x - (P1.x + P2.x)/2) + (P1.y + P2.y)/2
    -- LP2: y = -1/m2 *(x - (P2.x + P3.x)/2) + (P2.y + P3.y)/2
    --
    -- These two lines intersect at the centre, solving for the two equations
    -- LP1 and LP2 gives:
    --      m1*m2*(P1.y-P3.y) + m2*(P1.x+P2.x) - m1*(P2.x+P3.x)
    -- xc = ---------------------------------------------------
    --                        2*(m2-m1)

    LOCAL
     P1 : point  := a.start_point;
     P2 : point  := a.end_point;
     P3 : point  := a.through_point;
     m1 : NUMBER := (P2.y_coordinate - P1.y_coordinate)/
                    (P2.x_coordinate - P1.x_coordinate);
     m2 : NUMBER := (P3.y_coordinate - P2.y_coordinate)/
                    (P3.x_coordinate - P2.x_coordinate);
     xc, yc : NUMBER;
    END_LOCAL;
    
    xc := (m1 * m2 * (P1.y_coordinate - P3.y_coordinate) + 
           m2 * (P1.x_coordinate + P2.x_coordinate) -
           m1 * (P2.x_coordinate + P3.x_coordinate)) / 2 * (m2 - m1);

    yc := (xc - (P1.x_coordinate + P2.x_coordinate)/2)/(-m1) +
          (P1.y_coordinate + P2.y_coordinate) / 2;

    IF is_typeof(xc, 'INTEGER') AND is_typeof(yc, 'INTEGER') THEN
       RETURN(integer_point(xc DIV 1,yc DIV 1));
    ELSE
       RETURN(number_point(xc,yc));
    END_IF;
  END_FUNCTION;

  FUNCTION circle_and_circle_intersection
    (c1 : point; r1 : distance_value; 
     c2 : point; r2 : distance_value) : SET OF point;

    -- Let p1 and p2 be the points of intersection between two circles.  A
    -- triangle is formed between points c1, c2 and p1.
    -- 
    -- Given three sides of this triangle a, b and c where
    --   a = r1
    --   b = r2
    --          -------------------------------
    --         /             2                2
    --   c = \/ (c2.x - c1.x)  + (c2.y - c1.y)
    --
    -- so that s is half the perimeter of the triangle
    --       1
    --   s = - * (a + b + c)
    --       2
    --
    -- and the area of the triangle in terms of its sides is
    --             ---------------------
    --            /(s - a)(s - b)(s - c)
    --   area =  / ---------------------
    --         \/            s
    --
    -- Let the angle between side a and side c be M, then
    --   (s - b) * tan (M/2) = area/s
    --
    -- Let the angle of the line formed between c1 and c2 be N, then
    --            c2.y - c1.y
    --   N = ATAN(-----------)
    --            c2.x - x1.x
    --
    -- Finally, the coordinates of p1 are:
    --   p1.x = c1.x + r1 * cos(M + N)
    --   p1.y = c1.y + r1 * sin(M + N)
    --
    -- and the coordinates of p1 and p2 are:
    --   p2.x = c1.x + r2 * cos(M - N)
    --   p2.y = c1.y + r2 * sin(M - N)

    LOCAL
      a, b, c, s, area, x, y, M, N : NUMBER;
      p : SET OF point := [];
    END_LOCAL;

    a := r1;
    b := r2;
    c := SQRT((c2.x_coordinate - c1.x_coordinate)**2 +
              (c2.y_coordinate - c1.y_coordinate)**2);

    N := ATAN((c2.y_coordinate - c1.y_coordinate),
              (c2.x_coordinate - c1.x_coordinate));

    IF c = (a + b) THEN
       -- The two circles are touching, hence only one intersection point.
       x := c1.x_coordinate + r1 * cos(N);
       y := c1.y_coordinate + r1 * sin(N);
       IF is_typeof(x, 'INTEGER') AND is_typeof(y, 'INTEGER') THEN
          p[1] := integer_point(x DIV 1,y DIV 1);
       ELSE
          p[1] := number_point(x,y);
       END_IF;
    ELSE
    IF c < (a + b) THEN
       -- The two circles intersects at two points.
       s := (a + b + c)/2;
       area := SQRT((s - a) * (s - b) * (s - c)/s);
       M := 2 * ATAN((area/s),(s - b));

       x := c1.x_coordinate + r1 * cos(M + N);
       y := c1.y_coordinate + r1 * sin(M + N);
       IF is_typeof(x, 'INTEGER') AND is_typeof(y, 'INTEGER') THEN
          p[1] := integer_point(x DIV 1,y DIV 1);
       ELSE
          p[1] := number_point(x,y);
       END_IF;

       x := c1.x_coordinate + r2 * cos(M - N);
       y := c1.y_coordinate + r2 * sin(M - N);
       IF is_typeof(x, 'INTEGER') AND is_typeof(y, 'INTEGER') THEN
          p[2] := integer_point(x DIV 1,y DIV 1);
       ELSE
          p[2] := number_point(x,y);
       END_IF;
    END_IF;
    END_IF;
    RETURN(p);
  END_FUNCTION;

Description:

The circle_and_circle_intersection function returns the set of intersection points of two circles. A circle is specified by a centre point and its radius. The function returns an empty set if the two circles do not intersect.

circle_and_line_intersection

  FUNCTION circle_and_line_intersection
    (centre : point; radius : distance_value; ln : line_segment) : SET OF point;

    -- The formula of a line through p1 and p2 is represented as below:
    --   y = mx + q 
    --       where m = (p2.y - p1.y)/(p2.x - p1.x) and q = p1.y - m*p1.x
    --
    -- The formula of a circle with centre (u,v) and radius r is represented as
    -- below:
    --         2          2    2
    --  (x - u)  + (y - v)  = r
    --
    -- Hence writing y = mx + q in the equation to the circle, the abscissae of
    -- the points of intersection will be given by:
    --
    --        2   2                      2          2    2
    --  (1 + m  )x  + 2(mq - mv - u)x + u  + (q - v)  - r  = 0
    --
    -- This quadratic equation has real, equal or imaginary roots according as
    --   2                                                   2
    --  b - 4ac is positive, zero or negative where a = 1 + m
    --                                              b = 2(mq - mv - u)
    --                                                   2          2    2
    --                                              c = u  + (q - v)  - r
    --
    --     2
    -- If b - 4ac < 0, then the line does not intersect the circle.
    --
    --     2
    -- If b - 4ac = 0, then the line meets the circle in one point.  That is,
    -- the line is the tangent to the circle.  The x coordinate of this point
    -- is: 
    -- 
    --  x1 = - b / 2a
    --
    --     2
    -- If b - 4ac > 0, then the line meets the circle in two points.  The x
    -- coordinates of these two points are:
    --                -                    
    --  x1 = (- b + \/d ) / 2a              
    --                -                     2
    --  x2 = (- b - \/d ) / 2a  where d =  b - 4ac

    LOCAL
      a, b, c, d, m, q, u, v, x, y : NUMBER;
      p : SET OF point := [];
    END_LOCAL;

    u := centre.x_coordinate;
    v := centre.y_coordinate;

    m := (ln.end_point.y_coordinate - ln.start_point.y_coordinate) /
         (ln.end_point.x_coordinate - ln.start_point.x_coordinate);

    q := ln.start_point.y_coordinate - m * ln.start_point.x_coordinate;

    a := 1 + m**2;
    b := 2 * (m*q - m*v - u);
    c := u**2 + (q - v)**2 - radius**2;

    d := b**2 - 4 * a * c;

    IF d = 0 THEN
       x := -b / 2 * a;
       y := m * x + q;
       IF is_typeof(x, 'INTEGER') AND is_typeof(y, 'INTEGER') THEN
          p[1] := integer_point(x DIV 1,y DIV 1);
       ELSE
          p[1] := number_point(x,y);
       END_IF;
    ELSE
    IF d > 0 THEN
       x := (-b + SQRT(d)) / 2 * a;
       y := m * x + q;
       IF is_typeof(x, 'INTEGER') AND is_typeof(y, 'INTEGER') THEN
          p[1] := integer_point(x DIV 1,y DIV 1);
       ELSE
          p[1] := number_point(x,y);
       END_IF;

       x := (-b - SQRT(d)) / 2 * a;
       y := m * x + q;
       IF is_typeof(x, 'INTEGER') AND is_typeof(y, 'INTEGER') THEN
          p[2] := integer_point(x DIV 1,y DIV 1);
       ELSE
          p[2] := number_point(x,y);
       END_IF;
    END_IF;
    END_IF;
    RETURN(p);
  END_FUNCTION;

Description:

The circle_and_line_intersection function returns the set of intersection points of a circle and a line. A circle is specified by a centre point and its radius. The function returns an empty set if the circle and the line do not intersect.
