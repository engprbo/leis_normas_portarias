schematic_frame_model

Description:

The schematic_frame_model schema describes the structural connectivity and the implementation information of a frame in a schematic view. Different types of frame implementations are defined, including schematic_for_frame_implementation, schematic_if_frame_implementation and schematic_otherwise_frame_implementation. The structural connectivity information includes schematic_frame_bus and schematic_frame_net.

schematic_frame_implementation

  ENTITY schematic_frame_implementation
      ABSTRACT SUPERTYPE OF (ONEOF(schematic_for_frame_implementation,
                                   schematic_if_frame_implementation,
                                   schematic_otherwise_frame_implementation));
    busses                        : OPTIONAL SET [1:?] OF schematic_frame_bus;
    cell_property_displays        : OPTIONAL SET [1:?] OF property_display;
    cluster_property_displays     : OPTIONAL SET [1:?] OF property_display;
    comment_graphics_info         : OPTIONAL SET [1:?] OF comment_graphics;
    global_port_representations   : OPTIONAL SET [1:?] OF
                                    schematic_global_port_representation; 
    implemented_frame             : frame; -- reference
    instance_implementations      : OPTIONAL SET [1:?] OF 
                                    schematic_instance_implementation; 
    master_port_representations   : OPTIONAL SET [1:?] OF
                                    schematic_master_port_representation; 
    master_port_bundle_representations 
                                  : OPTIONAL SET [1:?] OF
                                    schematic_master_port_bundle_representation;
    nested_frame_implementations  : OPTIONAL SET [1:?] OF 
                                    schematic_frame_implementation;
    nets                          : OPTIONAL SET [1:?] OF schematic_frame_net;
    offpageconnectors             : OPTIONAL SET [1:?] OF
                                    schematic_offpageconnector; 
    onpageconnectors              : OPTIONAL SET [1:?] OF
                                    schematic_onpageconnector;  
    rippers                       : OPTIONAL SET [1:?] OF schematic_ripper;
    transform_position            : transform;
    view_property_displays        : OPTIONAL SET [1:?] OF property_display;
  DERIVE
    containing_page : page -- reference
      := usedin_one_role(SELF, 
         'schematic_implementation_model.page.frame_implementations');
  UNIQUE
    valid_implemented_frame :
      (* A frame can only be implemented on a page once per schematic. *)
      implemented_frame;
  WHERE
    containment_constraint :
      (* A schematic_frame_implementation belongs to one page. *)
      belongs_to_one(SELF,
        'schematic_implementation_model.page.frame_implementations'); 

    valid_global_port_representations :
      (* The global_port_representations must only reference the global_ports
         or global_port_bundles which are defined in the containing edif. *)
      not_exists(QUERY(globalPortRepresentation <* global_port_representations |
        NOT ((globalPortRepresentation.implemented_global_port IN 
              containing_page.containing_view.containing_cluster
              .containing_cell.containing_library.containing_edif.global_ports)
             OR
             (globalPortRepresentation.implemented_global_port IN 
              containing_page.containing_view.containing_cluster.containing_cell
              .containing_library.containing_edif.global_port_bundles))));

    valid_master_port_representations :
      (* The master_port_representations must only reference the master_ports
         which are defined in the interface of the containing cluster. *) 
      not_exists(QUERY(portRepresentation <* master_port_representations |
        NOT (portRepresentation.implemented_master_port
             IN containing_page.containing_view.interface.ports)));

    valid_master_port_bundle_representations :
      (* The master_port_bundle_representations must only reference the
         master_port_bundles which are defined in the interface of the
         containing cluster. *)  
      not_exists(QUERY(portBundleRepresentation 
        <* master_port_bundle_representations |
        NOT (portBundleRepresentation.implemented_master_port_bundle
             IN containing_page.containing_view.interface.port_bundles)));

    valid_instance_implementations :
      (* The implemented instances are defined in the implemented_frame. *)
      not_exists(QUERY(implementation <* instance_implementations |
       NOT (implementation.implemented_instance 
            IN implemented_frame.instances)));

    valid_nested_frame_implementations :
      (* Frames implemented in the nested_frame_implementations must be defined
         in the implemented_frame. *)
      not_exists(QUERY(nestedImplementation <* nested_frame_implementations |
        NOT (nestedImplementation.implemented_frame 
            IN implemented_frame.nested_frames)));  

    valid_cell_property_displays :
      (* The displayed properties are defined in the containing cell. *)
      not_exists(QUERY(cellPropertyDisplay <* cell_property_displays |
        NOT (cellPropertyDisplay.displayed_property
             IN containing_page.containing_view.containing_cluster
                .containing_cell.properties)));

    valid_cluster_property_displays :
      (* The displayed properties are defined in the containing
         internal_cluster. *) 
      not_exists(QUERY(clusterPropertyDisplay <* cluster_property_displays |
        NOT (clusterPropertyDisplay.displayed_property IN 
             containing_page.containing_view.containing_cluster.properties)));

    valid_view_property_displays :
      (* The displayed properties are defined in the containing
         internal_schematic_view. *) 
      not_exists(QUERY(viewPropertyDisplay <* view_property_displays |
        NOT (viewPropertyDisplay.displayed_property
             IN containing_page.containing_view.properties)));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. A frame implementation cannot have cell_property_displays if its
            containing cell does not define any properties.
         2. A frame implementation cannot have cluster_property_displays if its
            containing cluster does not define any properties.
         3. A frame implementation cannot have global_port_representations if
            its containing edif does not define any global ports.
         4. A frame implementation cannot have instance_implementations if its
            implemented frame does not define any instances.
         5. A frame implementation cannot have master_port_representations if
            its containing cluster does not define any master_ports.
         6. A frame implementation cannot have
            master_port_bundle_representations if its containing cluster does
            not define any master_port_bundles. 
         7. A frame implementation cannot have nested_frame_implementations if
            its implemented frame does not define any nested frames. *)
      (EXISTS(containing_page.containing_view.containing_cluster
              .containing_cell.properties) OR
         NOT EXISTS(cell_property_displays)) 
       AND
      (EXISTS(containing_page.containing_view.containing_cluster.properties) OR
         NOT EXISTS(cluster_property_displays))
       AND
      (EXISTS(containing_page.containing_view.containing_cluster
              .containing_cell.containing_library.containing_edif.global_ports)
         OR EXISTS(containing_page.containing_view.containing_cluster
                   .containing_cell.containing_library.containing_edif
                   .global_port_bundles) OR
         NOT EXISTS(global_port_representations))
       AND
      (EXISTS(implemented_frame.instances) OR
         NOT EXISTS(instance_implementations))
       AND
      (EXISTS(containing_page.containing_view.interface.ports) OR
         NOT EXISTS(master_port_representations))
       AND
      (EXISTS(containing_page.containing_view.interface.port_bundles) OR
         NOT EXISTS(master_port_bundle_representations))
       AND
      (EXISTS(implemented_frame.nested_frames) OR
         NOT EXISTS(nested_frame_implementations));
  END_ENTITY;

      EXISTS(instantiated_template.properties) OR
        NOT EXISTS(overriding_template_properties) AND
        NOT EXISTS(overriding_template_property_displays);

schematic_frame_generic_net

  ENTITY schematic_frame_generic_net
      ABSTRACT SUPERTYPE OF (ONEOF(schematic_frame_net, 
                                   schematic_frame_sub_net));
    associated_signal                  : frame_signal;         -- reference
    criticality                        : integer_token;                 
    delays                             : OPTIONAL SET [1:?] OF
                                         interconnect_delay;  
    document                           : OPTIONAL SET [1:?] OF documentation; 
    implementation                     : OPTIONAL
                                         schematic_interconnect_implementation; 
    joined_global_ports                : OPTIONAL SET [1:?] OF -- reference
                                         global_port;
    joined_global_port_representations  
                                       : OPTIONAL SET [1:?] OF -- reference
                                         schematic_global_port_representation;
    joined_instance_ports              : OPTIONAL SET [1:?] OF -- reference
                                         instance_port;
    joined_instance_port_representations
                                       : OPTIONAL SET [1:?] OF
                                         schematic_instance_port_representation;
    joined_master_ports                : OPTIONAL SET [1:?] OF -- reference
                                         master_port; 
    joined_master_port_representations  
                                       : OPTIONAL SET [1:?] OF -- reference
                                         schematic_master_port_representation; 
    joined_junctions                   : OPTIONAL SET [1:?] OF -- reference
                                         schematic_net_junction;
    joined_offpageconnectors           : OPTIONAL SET [1:?] OF -- reference
                                         schematic_offpageconnector;
    joined_onpageconnectors            : OPTIONAL SET [1:?] OF -- reference
                                         schematic_onpageconnector;
    joined_ripper_hotspots             : OPTIONAL SET [1:?] OF -- reference
                                         ripper_hotspot;
    joined_terminators                 : OPTIONAL SET [1:?] OF -- reference
                                         schematic_interconnect_terminator;
    name                               : OPTIONAL name_information;
    properties                         : OPTIONAL SET [1:?] OF property;
    sub_nets                           : OPTIONAL SET [1:?] OF
                                         schematic_frame_sub_net; 
  WHERE
    valid_sub_nets :
      (* No two schematic_frame_sub_nets may refer to the same set of ports and
         instantiatable schematic objects. *)

    not_exists(QUERY(net1 <* sub_nets |
      there_exists(QUERY(net2 <* sub_nets - net1 |
        (net1.joined_global_ports = net2.joined_global_ports) 
    AND (net1.joined_global_port_representations =
         net2.joined_global_port_representations)
    AND (net1.joined_instance_ports = net2.joined_instance_ports) 
    AND (net1.joined_instance_port_representations =
         net2.joined_instance_port_representations)
    AND (net1.joined_master_ports = net2.joined_master_ports) 
    AND (net1.joined_master_port_representations = 
         net2.joined_master_port_representations)
    AND (net1.joined_junctions = net2.joined_junctions)
    AND (net1.joined_offpageconnectors = net2.joined_offpageconnectors)
    AND (net1.joined_onpageconnectors = net2.joined_onpageconnectors)
    AND (net1.joined_ripper_hotspots = net2.joined_ripper_hotspots)
    AND (net1.joined_terminators = net2.joined_terminators)))));

    valid_global_port_representations :
      (* The joined global_port_representations reference global_ports which
         are joined by the interconnect. *)
      not_exists(QUERY(globalPortRepresentation
         <* joined_global_port_representations |
         NOT (globalPortRepresentation.implemented_global_port
              IN joined_global_ports)));

    valid_instance_port_representations :
      (* The joined instance_port_representations reference instance_ports
         which are joined by the interconnect. *)
      not_exists(QUERY(instancePortRepresentation
         <* joined_instance_port_representations |
         NOT (instancePortRepresentation.referenced_instance_port
              IN joined_instance_ports)));

    valid_master_port_representations :
      (* The joined master_port_representations reference master_ports which
         are joined by the interconnect. *)
      not_exists(QUERY(masterPortRepresentation
         <* joined_master_port_representations |
         NOT (masterPortRepresentation.implemented_master_port
              IN joined_master_ports)));

    valid_implementation :
      (* Implementation information may only be associated with a
         schematic_frame_generic_net when its sub_nets have no implementation
         information. *)
      NOT (EXISTS(implementation)
           AND
           there_exists(QUERY(subNet <* sub_nets |
             EXISTS(subNet\schematic_frame_generic_net.implementation))));

    valid_optional :
      (* The dependencies of these optional attributes are stated as follows:
         1. A net cannot have joined_global_port_representations if it does not
            join any global ports. 
         2. A net cannot have joined_instance_port_representations if it does
            not join any instance ports.
         3. A net cannot have joined_master_port_representations if it does not
            join any master_ports. *)

      (EXISTS(joined_global_ports) OR 
         NOT EXISTS(joined_global_port_representations))
      AND
      (EXISTS(joined_instance_ports) OR
         NOT EXISTS(joined_instance_port_representations))
      AND
      (EXISTS(joined_master_ports) OR
         NOT EXISTS(joined_master_port_representations));
  END_ENTITY;

Description:

 LEVEL 1
 =======
A schematic_frame_generic_net can be either a schematic_frame_net or a schematic_frame_sub_net. A schematic_frame_generic_net is used in the model because of the similarity between a schematic_frame_net and a schematic_frame_sub_net. A schematic_frame_net may be divided into schematic_frame_sub_nets and a schematic_frame_sub_net itself may also be divided into schematic_frame_sub_nets. No two schematic_frame_sub_nets at a given level may join to the same set of ports and instantiatable schematic objects.

Used by:

schematic_frame_sub_net schematic_interconnect_implementation schematic_net_junction

schematic_frame_net

  ENTITY schematic_frame_net
      SUBTYPE OF (schematic_frame_generic_net);
    junctions            : OPTIONAL SET [1:?] OF schematic_net_junction;
    schematic_wire_style : OPTIONAL schematic_wire_style;
    terminators          : OPTIONAL SET [1:?] OF
                           schematic_interconnect_terminator; 
  DERIVE
   containing_frame_implementation : schematic_frame_implementation -- reference
     := usedin_one_role(SELF,
        'schematic_frame_model.schematic_frame_implementation.nets'); 
  WHERE
    containment_constraint :
      (* A schematic_frame_net belongs to one schematic_frame_implementation. *)
      belongs_to_one(SELF, 
        'schematic_frame_model.schematic_frame_implementation.nets');

    valid_associated_signal :
      (* The frame_signal associated with a schematic_frame_net is defined in
         the implemented frame of the containing
         schematic_frame_implementation.*) 
      associated_signal IN
        containing_frame_implementation.implemented_frame.frame_signals; 

    valid_joined_ports : 
      (* The ports joined by a schematic_frame_net must be a subset of the
         connectivity of the associated signal of the schematic_frame_net. *)

        (joined_global_ports <= associated_signal.joined_global_ports)
    AND (joined_instance_ports <= associated_signal.joined_instance_ports)
    AND (joined_master_ports <= associated_signal.joined_master_ports);

    valid_joined_instance_ports :
      (* The joined instance_ports are defined in instances which are
         implemented in the containing schematic_frame_implementation. *)
      not_exists(QUERY(instancePort <* joined_instance_ports |
        NOT (usedin_one_role(instancePort.referenced_instance,
                             specified_role('schematic_implementation_model',
                                            'schematic_instance_implementation',
                                            'implemented_instance'))
             IN containing_frame_implementation.instance_implementations)));

    valid_joined_instantiatable_objects :
      (* All joined schematic_global_port_representations,
         schematic_master_port_representations, schematic_offpageconnectors,
         schematic_onpageconnectors and ripper_hotspots are defined in
         the containing schematic_frame_implementation. *)

        (joined_global_port_representations
         <= containing_frame_implementation.global_port_representations)
    AND (joined_master_port_representations
         <= containing_frame_implementation.master_port_representations)
    AND (joined_offpageconnectors
         <= containing_frame_implementation.offpageconnectors)
    AND (joined_onpageconnectors
         <= containing_frame_implementation.onpageconnectors)
    AND (not_exists(QUERY(hotspot <* joined_ripper_hotspots |
        NOT (hotspot.containing_ripper 
             IN containing_frame_implementation.rippers))));

    valid_joined_junctions :
      (* The joined junctions must be defined in the schematic_frame_net. *)
      joined_junctions <= junctions;

    valid_joined_terminators :
      (* The joined terminators must be defined in the schematic_frame_net. *)
      joined_terminators <= terminators;

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. A net cannot have joined_instance_ports if its containing frame
            implementation does not define any instance implementations and its
            associated signal does not join any instance_ports or
            instance_member_port_members. 
         2. A net cannot have joined_global_ports if its associated signal does
            not join any global ports.
         3. A net cannot have joined_master_ports if its associated signal does
            not join any master_ports or master_port_members. 
         4. A net cannot have joined_global_port_representations if its
            containing frame implementation does not define any global port
            representations. 
         5. A net cannot have joined_master_port_representations if its
            containing frame implementation does not define any master port
            representations. 
         6. A net cannot have joined_offpageconnectors if its containing frame
            implementation does not define any offpageconnectors.
         7. A net cannot have joined_onpageconnectors if its containing frame
            implementation does not define any onpageconnectors.
         8. A net cannot have joined_ripper_hotspots if its containing frame
            implementation does not define any rippers.
         9. A net cannot have joined_junctions if it does not define any
            junctions. 
         10.A net cannot have joined_terminators if it does not define any
            terminators. *)

      (EXISTS(containing_frame_implementation.instance_implementations) AND
         EXISTS(associated_signal.joined_instance_ports) OR
         EXISTS(associated_signal.joined_instance_member_port_members) OR
         NOT EXISTS(joined_instance_ports))
      AND
      (EXISTS(associated_signal.joined_global_ports) OR
         NOT EXISTS(joined_global_ports))
      AND
      (EXISTS(associated_signal.joined_master_ports) OR
         EXISTS(associated_signal.joined_master_port_members) OR
         NOT EXISTS(joined_master_ports))
      AND
      (EXISTS(containing_frame_implementation.global_port_representations) OR
         NOT EXISTS(joined_global_port_representations))
      AND
      (EXISTS(containing_frame_implementation.master_port_representations) OR
         NOT EXISTS(joined_master_port_representations))
      AND
      (EXISTS(containing_frame_implementation.offpageconnectors) OR
         NOT EXISTS(joined_offpageconnectors))
      AND
      (EXISTS(containing_frame_implementation.onpageconnectors) OR
         NOT EXISTS(joined_onpageconnectors))
      AND
      (EXISTS(containing_frame_implementation.rippers) OR
         NOT EXISTS(joined_ripper_hotspots))
      AND
      (EXISTS(junctions) OR NOT EXISTS(joined_junctions))
      AND
      (EXISTS(terminators) OR NOT EXISTS(joined_terminators));
  END_ENTITY;

Description:

 LEVEL 1
 =======
A schematic_frame_net is defined within a schematic_frame_implementation which describes the structural connectivity information of a frame in a schematic view. Its associated frame_signal must be defined in the implemented frame of the containing schematic_frame_implementation. The ports joined by a schematic_frame_net must be a subset of the ones joined by its associated frame_signal. Also, all joined instantiatable schematic objects must be defined in the containing schematic_frame_implementation.

Used by:

schematic_frame_implementation

  ENTITY schematic_frame_sub_net
      SUBTYPE OF (schematic_frame_generic_net);
  DERIVE
    containing_generic_net : schematic_frame_generic_net -- reference
      := usedin_one_role(SELF,
         'schematic_frame_model.schematic_frame_sub_net.sub_nets'); 
  WHERE
    containment_constraint :

    AND (joined_instance_port_bundles

         schematic_master_port_representations,

            if its containing bus/bus-slice/sub-bus does not join any instance
            port bundle representations. 
         11.A sub-bus cannot have joined_master_port_representations if its
            containing bus/bus-slice/sub-bus does not join any master port

         12.A sub-bus cannot have joined_master_port_bundle_representations if
            its containing bus/bus-slice/sub-bus does not join any master port

         13.A sub-bus cannot have joined_junctions if its containing

      (EXISTS(containing_generic_bus.joined_global_ports) OR

 LEVEL 1
 =======
A schematic_frame_sub_bus has the same size as the schematic_frame_generic_bus immediately containing it. Its associated frame_signal_group is the same as that of its immediately containing schematic_frame_generic_bus. A schematic_frame_sub_bus must only join to ports and instantiatable schematic objects which are already joined by its immediately containing schematic_frame_generic_bus.

containing_bus_of_schematic_frame_generic_bus

  FUNCTION containing_bus_of_schematic_frame_generic_bus
     (busStructure : schematic_frame_generic_bus) : schematic_frame_bus;
    LOCAL
      busType : SET OF STRING := [ ];
    END_LOCAL;

    REPEAT UNTIL ('schematic_frame_model.schematic_frame_bus' 
                  IN busType);
      busType := TYPEOF(busStructure);
      IF 'schematic_frame_model.schematic_frame_bus_slice' IN busType THEN
         busStructure := usedin_one_role(busStructure,
         'schematic_frame_model.schematic_frame_generic_bus.bus_slices');
      ELSE
      IF 'schematic_frame_model.schematic_frame_sub_bus' IN busType THEN
         busStructure := usedin_one_role(busStructure,
         'schematic_frame_model.schematic_frame_generic_bus.sub_busses');
      END_IF;
      END_IF;
    END_REPEAT;

    RETURN(busStructure);
  END_FUNCTION;
