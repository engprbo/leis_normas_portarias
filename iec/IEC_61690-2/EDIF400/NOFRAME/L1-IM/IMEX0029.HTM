pcb_sub_layout_model

Description:

The pcb_sub_layout_model schema is used to describes various types of sub_layout_definitions which can be instanced on many different bare boards. For example, a footprint_definition may consist of a number of toeprint_instance, where each toeprint_instance references a toeprint_definition. The footprint_definition can then be instanced within a bare_board_definition.

sub_layout_definition

  ENTITY sub_layout_definition
      ABSTRACT SUPERTYPE OF (ONEOF (bare_board_definition,
                                    test_coupon_definition,
                                    footprint_definition,
                                    toeprint_definition,
                                    non_specific_sub_layout_definition,
                                    padstack_definition)
                             AND
                             ONEOF (external_sub_layout_definition,
                                    internal_sub_layout_definition));
    document          : OPTIONAL SET [1:?] OF documentation;
    is_implemented_in : bare_board_technology; -- reference
    name              : OPTIONAL name_information;
    physical_nets     : OPTIONAL SET [1:?] OF physical_net;
    previous_version  : OPTIONAL sub_layout_definition; -- reference
    properties        : OPTIONAL SET [1:?] OF property;
    units             : pcb_layout_units;
    written_status    : OPTIONAL SET [1:?] OF written;
  WHERE
    valid_existence :
      (* A sub_layout_definition belongs to one library. *)
      belongs_to_one(SELF,'library_model.library.sub_layout_definitions');

    valid_previous_version :
      (* A sub_layout_definition cannot be a new version of itself, either
         directly or indirectly. *)
      non_recursive_previous_version(SELF, []);

    valid_written_in_previous_version :
      (* The earliest timestamp of a sub_layout_definition is later than the
         latest timestamp of its previous version. *)  
      valid_timestamp_in_previous_version(written_status,
                                          previous_version.written_status);
  END_ENTITY;

Description:

A sub_layout_definition object is a description of a collection of layout items, which can later be instanced into another sub-layout or onto a bare board. An example of a sub-layout is a footprint (a collection of layout items which is specifically reserved for connection to the pins of a component).

Used by:

library physical_net sub_layout_definition sub_layout_instance

  ENTITY internal_sub_layout_definition
      ABSTRACT SUPERTYPE OF (ONEOF (internal_bare_board_definition,
                                    internal_test_coupon_definition, 
                                    internal_footprint_definition, 
                                    internal_toeprint_definition, 
                                    internal_non_specific_sub_layout_definition,
                                    internal_padstack_definition))
      SUBTYPE OF (sub_layout_definition);
    SELF\sub_layout_definition.physical_nets -- retype
                          : OPTIONAL SET [1:?] OF internal_physical_net;
    layout_features       : OPTIONAL SET [1:?] OF layout_feature;
    associated_text       : OPTIONAL SET [1:?] OF layout_text;
    drawing_information   : OPTIONAL SET [1:?] OF drawing_element;
    name_fabrication      : OPTIONAL SET [1:?] OF fabricate_text_appearance;
    property_fabrications : OPTIONAL SET [1:?] OF property_fabrication;
  WHERE
    valid_existence :
      (* An internal_sub_layout_definition belongs to one internal_library. *)
      belongs_to_one(SELF,
        'library_model.internal_library.sub_layout_definitions');

    valid_property_fabrications :
      (* The fabricated properties are defined in the sub_layout_definition. *)
      not_exists(QUERY(propertyFabrication <* property_fabrications |
        NOT (propertyFabrication.fabricated_property 
               IN SELF\sub_layout_definition.properties)));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. A sub-layout cannot have property_fabrications if it does not
            define any properties. *)
      EXISTS(SELF\sub_layout_definition.properties) 
        OR NOT EXISTS(property_fabrications);
  END_ENTITY;

bare_board_definition

  ENTITY bare_board_definition
      ABSTRACT SUPERTYPE OF (ONEOF (external_bare_board_definition,
                                    internal_bare_board_definition))
      SUBTYPE OF (sub_layout_definition);
    SELF\sub_layout_definition.physical_nets     
        : OPTIONAL SET [1:?] OF bare_board_physical_net; -- retype
    SELF\sub_layout_definition.previous_version 
        : OPTIONAL bare_board_definition;                -- retype
    sub_layouts   : OPTIONAL SET [1:?] OF sub_layout_instance;
    embedded_dies : OPTIONAL SET [1:?] OF embedded_die; -- chips-first
  WHERE
    valid_sub_layouts :
      (* The bare_board_technology of the sub_layout_instances must be the same
         as the one used by the bare_board_definition. *)
      not_exists(QUERY(subLayoutInstance <* sub_layouts |
        subLayoutInstance.is_defined_by.is_implemented_in
        :<>: SELF\sub_layout_definition.is_implemented_in));
  END_ENTITY;

Description:

A bare board is a composition of laminated sheets of conductive material, separated by insulation sheets, and usually interconnected at several places. The bare board is used in the assembly process, to mount packaged (pre-fabricated) components onto, and appears in the bill of material of the assembled board.

A bare_board_definition is a description of a re-usable bare board.

Used by:

internal_bare_board_definition

  ENTITY internal_bare_board_definition
      SUBTYPE OF (internal_sub_layout_definition,
                  bare_board_definition);
    SELF\sub_layout_definition.physical_nets -- retype  
        : OPTIONAL SET [1:?] OF internal_bare_board_physical_net;
    drawings                              : OPTIONAL SET [1:?] OF pcb_drawing;
    drawing_board_subsets                 : OPTIONAL SET [1:?] OF board_subset;
    outline                               : board_outline;
    probe_points                          : OPTIONAL SET [1:?] OF probe_point;
    embedded_die_placements               : OPTIONAL SET [1:?] OF 
                                            embedded_die_placement;
    sub_layout_placements                 : OPTIONAL SET [1:?] OF
                                            sub_layout_placement; 
    assembled_board_name_fabrication_slot : OPTIONAL SET [1:?] OF
                                            fabricate_text_appearance;
    constraint                            : OPTIONAL bare_board_constraint;
    cavities                              : OPTIONAL SET [1:?] OF
                                            bare_board_cavity; -- chips-first
  WHERE
    valid_embedded_die_placements :
      (* The placed dies must be defined in the bare_board_definition. *)
      not_exists(QUERY(embeddedDiePlacement <* embedded_die_placements |
        NOT (embeddedDiePlacement.placed_die 
               IN SELF\bare_board_definition.embedded_dies)));

    valid_sub_layout_placements :
      (* The placed sub_layout_instance must be defined in the
         bare_board_definition. *)
      not_exists(QUERY(subLayoutPlacement <* sub_layout_placements |
        NOT (subLayoutPlacement.placed_instance 
               IN SELF\bare_board_definition.sub_layouts)));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. A bare board cannot have embedded_die_placements if it does not
            define any embedded dies.
         2. A bare board cannot have sub_layout_placements if it does not
            define any sub-layouts. *)
      (EXISTS(SELF\bare_board_definition.embedded_dies) 
         OR NOT EXISTS(embedded_die_placements))
      AND
      (EXISTS(SELF\bare_board_definition.sub_layouts) 
         OR NOT EXISTS(sub_layout_placements));
  END_ENTITY;

Description:

A bare_board_cavity describes a cavity on a layer of a bare board in which dies are placed. A cavity consists one or more bare_board_cavity_wells which may or may not have a uniform depth.

The applied_layer attribute specifies the physical layer in which a cavity is formed so that bare dies are placed inside. The side attribute determines whether a cavity is formed on the top side or on the bottom side of the layer. The surface_area attribute specifies the area of the cavity on the substrate surface.

Used by:

bare_board_cavity_well internal_bare_board_definition

bare_board_cavity_well

  ENTITY bare_board_cavity_well;
    wall_angle         : angle_value;
    surface_area       : OPTIONAL closed_area_element;
    bottom_area        : OPTIONAL closed_area_element;
    die_placement_area : OPTIONAL closed_area_element;
    depth              : toleranced_distance;
  DERIVE
    containing_cavity : bare_board_cavity -- reference
      := usedin_one_role(SELF, 'pcb_sub_layout_model.bare_board_cavity.wells');
  WHERE
    valid_existence :
      (* A bare_board_cavity_well belongs to one bare_board_cavity. *)
       belongs_to_one(SELF, 'pcb_sub_layout_model.bare_board_cavity.wells');

    valid_depth :
      (* The depth of the well must be less than the thickness of the
         layer.  This constraint can be checked only if the layer is an
         internal_physical_layer for which the thickness of a layer is
         known. *)  
      NOT (is_typeof(containing_cavity.applied_layer, 
                     'pcb_technology_model.internal_physical_layer')
           AND
           EXISTS(containing_cavity
                  .applied_layer\internal_physical_layer.thickness)
           AND
           (containing_cavity
            .applied_layer\internal_physical_layer.thickness.range.min
            <= depth.range.max));
  END_ENTITY;

Description:

A bare_board_cavity_well describes a well inside a bare_board_cavity.

The wall_angle attribute specifies the angle of the well wall measured from the bottom of the well.

The surface_area attribute specifies the area of the well as seen from the layer surface. If it is not specified (e.g. when a cavity has only one well), uses the surface_area information from the bare_board_cavity.

The bottom_area attribute specifies the area of the bottom of the well as seen from the layer surface. It it is not specified (e.g. when the angle of the well wall is equal to 90 degrees), use the surface_area information from the well.

The die_placement_area attribute specifies the well area in which dies can be placed. If it is not specified, use the bottom_area information from the well.

Used by:

bare_board_cavity embedded_die_placement_region

embedded_die_placement

  ENTITY embedded_die_placement;
    placed_die           : embedded_die; -- reference
    placement            : embedded_die_placement_region;
    die_position         : unscaled_transform;
    mounting_template    : mountable_die_physical_template; -- reference
    referenced_footprint : footprint_instance; -- reference
    die_terminals        : OPTIONAL SET [1:?] OF embedded_die_terminal;
  WHERE 
    valid_existence :
      (* An embedded_die_placement belongs to one
         internal_bare_board_definition. *)  
      belongs_to_one(SELF,
        specified_role('pcb_sub_layout_model',
                       'internal_bare_board_definition',
                       'embedded_die_placements'));

    valid_mounting_side :
      (* If the mounting_template is an
         internal_mountable_die_physical_template and the die is placed inside
         a bare_board_cavity_well, then the mounting side of the template must
         be the same as the side of the cavity. *)
    NOT (
    is_typeof(mounting_template,
    'mcm_mountable_die_physical_model.internal_mountable_die_physical_template')
    AND
      is_typeof(placement, 'pcb_sub_layout_model.bare_board_cavity_well')
      AND
      (((mounting_template\internal_mountable_die_physical_template
         .side_of_mounting = mounting_side.top)
         AND
        (placement.containing_cavity.side
         = physical_layer_side.bottom_of_layer)) 
       OR
       ((mounting_template\internal_mountable_die_physical_template
         .side_of_mounting = mounting_side.bottom)
         AND
        (placement.containing_cavity.side
         = physical_layer_side.top_of_layer))));

    valid_mounting_template :
      (* The referenced mountable_die_physical_template must refer to the same
         die_physical_template as the embedded die_electrical_part. *)
    NOT (
    is_typeof(mounting_template,
    'mcm_mountable_die_physical_model.internal_mountable_die_physical_template')
    AND
      (placed_die.embedded_part.referenced_die_physical_template
       :<>:
       mounting_template\internal_mountable_die_physical_template
                         .referenced_die_physical_template));

    valid_referenced_footprint :
      (* The referenced footprint_instance must be within the containing
         internal_bare_board_definition. *)
      EXISTS(placed_die.containing_board.sub_layouts)
      AND
      (referenced_footprint 
         IN placed_die.containing_board.sub_layouts);
  END_ENTITY;

The placement attribute specifies a layer or a cavity on a layer in which the die is placed. The mounting_template attribute describes how a die is placed on the layer.

NOTE: The mounting_template uses the mountable_die_physical_template which describes various chip bonding technologies, such as wirebonds, TABs and solder-bumps.

Description:

A test_coupon_definition object is a description of a collection of layout items, which can later be instanced into another sub-layout or onto a bare board as a test coupon.

A test coupon is a collection of layout items which is used purely for testing purposes, and will typically consist of the narrowest possible track, tracks which are as close together as allowed by the design technology, the smallest allowable holes, and so on. This region of the bare board can be subjected to a series of (possibly destructive) tests. If this region fails the tests, then it is likely that the board as a whole has not been manufactured correctly. (If it passes the tests, it does no more than increase the tester's confidence in the state of the rest of the bare board).

The internal nets of a test coupon would normally have no intended functional impact on the printed circuit board.

Used by:

test_coupon_definition test_coupon_instance

non_recursive_non_specific_sub_layout

  FUNCTION non_recursive_non_specific_sub_layout 
    (referenced, current_level : SET OF non_specific_sub_layout_definition) 
    : BOOLEAN;
    LOCAL
      next_level : SET OF non_specific_sub_layout_definition := [];
      result     : BOOLEAN := TRUE;
      i          : INTEGER;
      j          : INTEGER;
      instances  : SET OF non_specific_sub_layout_instance;
    END_LOCAL;
    referenced := referenced + current_level;
    REPEAT i := LOINDEX (current_level) TO HIINDEX (current_level);
    (*
      For all non specific sub layout definitions in the current level, find the
      sub layout instances contained within them.
    *)
      instances := current_level [i].other_sub_layouts;
      REPEAT j := LOINDEX (instances) TO HIINDEX (instances);
        next_level := next_level + instances [j].is_defined_by;
      END_REPEAT;
    END_REPEAT;
    IF (referenced * next_level <> []) THEN
      result := FALSE;
    ELSE
      IF (next_level <> []) THEN
       result := non_recursive_non_specific_sub_layout (referenced, next_level);
      ELSE
       result := TRUE;
      END_IF;
    END_IF;
    RETURN (result);
  END_FUNCTION;

  ENTITY trace
      SUBTYPE OF (layout_feature);
    actual_geometry    : OPTIONAL fabricate_figure; -- open figure
    clearance_geometry : OPTIONAL SET [1:?] OF fabricate_figure; -- open figure
  DERIVE
     positive_layers : SET OF physical_layer
       := QUERY (layer <* referenced_technology.layers |
            (actual_geometry.contributes_to_fabrication *
             layer.positive_contributing_collectors) <> []);
  WHERE
    valid_actual_geometry :
      (* If actual_geometry exists, it must contribute positively to at least
         one physical_layer. *)
      NOT (EXISTS(actual_geometry) AND (positive_layers = []));

    valid_clearance_geometry_from_actual_geometry :
      (* If both actual_geometry and clearance_geometry exist, then the
         clearance_geometry must clear from at least one physical_layer 
         which the actual_geometry contributes to. *) 
      NOT (EXISTS(actual_geometry) AND EXISTS(clearance_geometry) AND
           there_exists (QUERY(cg <* clearance_geometry |
              not_exists (QUERY (layer <* positive_layers |
                cg.contributes_to_fabrication <=
                layer.negative_contributing_collectors)))));

    valid_clearance_geometry :
      (* If only clearance_geometry exists, then the clearance_geometry must
         clear from at least one physical_layer. *)
      NOT (EXISTS(clearance_geometry) AND NOT EXISTS(actual_geometry) AND
           there_exists (QUERY(cg <* clearance_geometry |
              not_exists (QUERY (layer <* referenced_technology.layers |
                cg.contributes_to_fabrication <=
                layer.negative_contributing_collectors)))));
  END_ENTITY;

  FUNCTION non_recursive_physical_net
    (parents, invalidAncestors : SET OF physical_net) : BOOLEAN;
    LOCAL
      i           : INTEGER;
      descendents : SET OF physical_net := [];
    END_LOCAL;

    REPEAT i := 1 TO SIZEOF(parents);
      descendents := descendents + parents[i].internal_nets;
    END_REPEAT;
    invalidAncestors := parents + invalidAncestors;
    RETURN(not_exists(descendents)
           XOR
           (not_exists(descendents * invalidAncestors)
            AND 
            non_recursive_physical_net(descendents, invalidAncestors)));
  END_FUNCTION;
