pcb_drawing_model

Description:

The pcb_drawing_model schema describes the objects which are used to specify drawing documentation for printed circuit boards and assemblies.

pcb_drawing

  ENTITY pcb_drawing;
    annotations                    : OPTIONAL SET [1:?] OF annotation_note;
    annotation_groups              : OPTIONAL SET [1:?] OF annotation_group;
    board_subset_placements        : OPTIONAL SET [1:?] OF
                                     board_subset_placement;
    border                         : OPTIONAL drawing_border;
    conforms_to                    : dimensioning_standard;
    datum_dimension_line_length    : length_value;
    default_dimension_units        : pcb_dimension_units;
    default_scale_factor           : scale_factor;
    dimension_end_geometries       : OPTIONAL SET [1:?] OF
                                     dimension_end_geometry; 
    dimensioning_information       : OPTIONAL SET [1:?] OF dimension;
    drawing_size                   : OPTIONAL string_token;
    drawing_units                  : pcb_drawing_units;
    external_dimension_line_length : length_value;
    name                           : OPTIONAL name_information;
    physical_paper_size            : OPTIONAL rectangle;
    projection_line_gap            : non_negative_distance_value;
    projection_line_overshoot      : non_negative_distance_value;
    properties                     : OPTIONAL SET [1:?] OF property;
    property_drawings              : OPTIONAL SET [1:?] OF property_drawing;
    status_of_written              : OPTIONAL SET [1:?] OF written;
    text_enclosure_clearance       : non_negative_distance_value;
    title_block                    : OPTIONAL drawing_title_block;
  DERIVE
    containing_assembled_board : assembled_board -- reference
      := usedin_one_role(SELF,
         'pcb_assembled_board_model.assembled_board.drawings');
    containing_bare_board : internal_bare_board_definition -- reference
      := usedin_one_role(SELF,
         'pcb_sub_layout_model.internal_bare_board_definition.drawings');
  WHERE
    valid_existence:
      (* A pcb_drawing must belong to one assembled_board or to one
         bare_board_definition. *)
      either_belongs_to_one(SELF,
        ['pcb_assembled_board_model.assembled_board.drawings',
         'pcb_sub_layout_model.internal_bare_board_definition.drawings']);

    valid_property_drawings :
      (* The drawn properties are defined in the drawing. *)
      not_exists(QUERY(propertyDrawing <* property_drawings |
        NOT (propertyDrawing.drawn_property IN properties)));

    valid_optional_attributes :
      (* The dependencies of these optional attributes are stated as follows:
         1. A drawing cannot have annotation_groups if it does not define any
            annotation_notes. 
         2. A drawing cannot have property_drawings if it does not define any
            properties.
         3. A drawing cannot have dimensioning_information if it does not
            define any board_subset_placements.  *)
      (EXISTS(annotations) OR NOT EXISTS(annotation_groups))
      AND
      (EXISTS(properties) OR NOT EXISTS(property_drawings))
      AND
      (EXISTS(board_subset_placements) OR NOT EXISTS(dimensioning_information));
  END_ENTITY;

board_subset_placement

  ENTITY board_subset_placement
    ABSTRACT SUPERTYPE OF(ONEOF(scaled_board_subset_placement,
                                unscaled_board_subset_placement));
    associated_figures                : OPTIONAL SET [1:?] OF drawing_figure;
    clipping_outline_drawing_defaults : -- reference
                                        OPTIONAL pcb_drawing_attribute_default;
    referenced_board_subset           : board_subset; -- reference
    position                          : uniform_scale_transform;
  DERIVE
    containing_drawing : pcb_drawing -- reference
      := usedin_one_role(SELF,
         'pcb_drawing_model.pcb_drawing.board_subset_placements');
  WHERE
    valid_existence:
      (* a board_subset_placement must belong to a pcb_drawing *)
      belongs_to_one(SELF,
        'pcb_drawing_model.pcb_drawing.board_subset_placements');

    valid_board_subset:
      (* The referenced board_subset is defined in the assembled_board or
         internal_bare_board_definition containing the pcb_drawing. *) 
    (EXISTS(containing_drawing.containing_assembled_board.drawing_board_subsets)
     AND
     (referenced_board_subset
        IN containing_drawing.containing_assembled_board.drawing_board_subsets))
    OR
    (EXISTS(containing_drawing.containing_bare_board.drawing_board_subsets)
     AND
     (referenced_board_subset
        IN containing_drawing.containing_bare_board.drawing_board_subsets));
  END_ENTITY;

scale_factor

  ENTITY scale_factor;
    scale_value    : scale_representation;
    display_string : string_token;
  WHERE
    valid_existence :
      (* A scale_factor belongs to one scaled_board_subset_placement. *)

        'pcb_drawing_model.scaled_board_subset_placement.scale');

      (* One factor must be 1 and the other must be greater than 1. *)

single_angular_dimension

  ENTITY single_angular_dimension
    SUBTYPE OF (angular_dimension);
    measure_item   : angular_dimension_item;
    arrow_geometry : dimension_end_geometry; -- reference
  WHERE
    valid_measure_item:
      (* The end line must not be parallel with the start line *)
      NOT parallel_dimensioned_line(SELF\angular_dimension.start_line,
                                    measure_item.end_line);

    valid_dimension :
      (* It cannot be used to dimension objects between two
         board_subset_extra_details, or between a board_subset_definition and a
         board_subset_extra_detail. *)
      (is_typeof(SELF\angular_dimension.start_line
                 .dimensioned_board.referenced_board_subset, 
                 'pcb_board_subset_model.board_subset_definition')
       AND
       is_typeof(measure_item.end_line
                 .dimensioned_board.referenced_board_subset, 
                 'pcb_board_subset_model.board_subset_definition'))
      OR
      (is_typeof(SELF\angular_dimension.start_line
                 .dimensioned_board.referenced_board_subset,
                 'pcb_board_subset_model.board_subset_extra_detail')
       AND
       is_typeof(measure_item.end_line
                 .dimensioned_board.referenced_board_subset,
                 'pcb_board_subset_model.board_subset_extra_detail')
       AND
       (SELF\angular_dimension.start_line
            .dimensioned_board.referenced_board_subset
        :=:
        measure_item.end_line.dimensioned_board.referenced_board_subset));

    valid_arrow_geometry :
      (* The referenced dimension_end_geometry must be defined in the
         containing pcb_drawing. *)
      EXISTS(SELF\dimension.containing_drawing.dimension_end_geometries)
      AND
      (arrow_geometry IN
         SELF\dimension.containing_drawing.dimension_end_geometries); 
  END_ENTITY;

multiple_angular_dimension

  ENTITY multiple_angular_dimension
    ABSTRACT SUPERTYPE OF(ONEOF(datum_angular_dimension,
                                chained_angular_dimension))
    SUBTYPE OF (angular_dimension);
    measure_items : LIST [2:?] OF UNIQUE angular_dimension_item;
  WHERE
    valid_measure_items:
      (* The end line must not be parallel with the start line. *)
      not_exists(QUERY(item <* measure_items |
        parallel_dimensioned_line(SELF\angular_dimension.start_line,
                                  item.end_line)));

    valid_dimension :
      (* It cannot be used to dimension objects between two
         board_subset_extra_details, or between a board_subset_definition and a
         board_subset_extra_detail. *)
      (is_typeof(SELF\angular_dimension.start_line
                 .dimensioned_board.referenced_board_subset,
                 'pcb_board_subset_model.board_subset_definition')
       AND
       not_exists(QUERY(item <* measure_items |
          NOT is_typeof(item.end_line
                        .dimensioned_board.referenced_board_subset,
                        'pcb_board_subset_model.board_subset_definition'))))
      OR
      (is_typeof(SELF\angular_dimension.start_line
                 .dimensioned_board.referenced_board_subset,
                 'pcb_board_subset_model.board_subset_extra_detail')
       AND
       not_exists(QUERY(item <* measure_items |
         is_typeof(item.end_line
                   .dimensioned_board.referenced_board_subset, 
                   'pcb_board_subset_model.board_subset_definition')
         OR
         (SELF\angular_dimension.start_line
              .dimensioned_board.referenced_board_subset
          :<>:
          item.end_line.dimensioned_board.referenced_board_subset))));
  END_ENTITY;

parallel_dimensioned_line

  FUNCTION parallel_dimensioned_line
    (startLine, endLine : board_subset_dimensioned_line) : BOOLEAN;
    LOCAL
      line1    : line_segment := startLine.dimensioned_figure.reference_line;
      line2    : line_segment := endLine.dimensioned_figure.reference_line;
      parallel : BOOLEAN;
    END_LOCAL;
    parallel := 
      ((line1.end_point.y_coordinate - line1.start_point.y_coordinate)
       * 
       (line2.end_point.x_coordinate - line2.start_point.x_coordinate))
      =
       ((line1.end_point.x_coordinate - line1.start_point.x_coordinate)
       * 
       (line2.end_point.y_coordinate - line2.start_point.y_coordinate));
    RETURN(parallel);
  END_FUNCTION;

valid_chained_angular_dimension

  FUNCTION valid_chained_angular_dimension
    (startLine    : board_subset_dimensioned_line;
     measurements : LIST OF angular_dimension_item) : BOOLEAN;
    LOCAL
      ln1, ln2 : line_segment;
      angle    : NUMBER := 0;
    END_LOCAL;

    -- Calculate the accumulated angle between all measured lines
    ln1 := startLine.dimensioned_figure.reference_line;
    ln2 := measurements[1].end_line.dimensioned_figure.reference_line;
    REPEAT i := 1 TO SIZEOF(measurements)-1;
      angle := angle + ABS(angle_of_line(ln2.start_point, ln2.end_point) -
                           angle_of_line(ln1.start_point, ln1.end_point));
      ln1 := measurements[i].end_line.dimensioned_figure.reference_line;
      ln2 := measurements[i+1].end_line.dimensioned_figure.reference_line;
    END_REPEAT;

    -- Calculate the angle between the start line and the end line of the
    -- last measurements.  
    ln1 := startLine.dimensioned_figure.reference_line;
    ln2 := measurements[SIZEOF(measurements)].end_line.dimensioned_figure
           .reference_line;

    -- If the angle between the start line and the end line of the last

  FUNCTION valid_chained_linear_dimension
    (startPt      : board_subset_dimensioned_point;
     measurements : LIST OF linear_dimension_item) : BOOLEAN;
    LOCAL
      pt1, pt2 : integer_point;
      distance : NUMBER := 0;
    END_LOCAL;

    -- Calculate the accumulated distance between all measured positions
    pt1 := startPt.dimensioned_figure.reference_point;
    pt2 := measurements[1].end_point.dimensioned_figure.reference_point;
    REPEAT i := 1 TO SIZEOF(measurements)-1;
      distance := distance + SQRT((pt2.x_coordinate-pt1.x_coordinate)**2 +
                                  (pt2.y_coordinate-pt1.y_coordinate)**2);
      pt1 := measurements[i].end_point.dimensioned_figure.reference_point;
      pt2 := measurements[i+1].end_point.dimensioned_figure.reference_point;
    END_REPEAT;

    -- Calculate the distance between the start point and the end point of the
    -- last measurements.  
    pt1 := startPt.dimensioned_figure.reference_point;
    pt2 := measurements[SIZEOF(measurements)].end_point.dimensioned_figure
           .reference_point;

    -- If the distance between the start point and the end point of the last
    -- measurements is equal to the accumlated distance between all measured
    -- positions, then the function returns true. 
    distance := distance - SQRT((pt2.x_coordinate-pt1.x_coordinate)**2 +
                                (pt2.y_coordinate-pt1.y_coordinate)**2);
    RETURN(distance = 0);
  END_FUNCTION;

  FUNCTION valid_datum_angular_dimension_centre_point
    (pt        : number_point;
     startLine : board_subset_dimensioned_line;
     items     : LIST OF angular_dimension_item) : BOOLEAN;

    LOCAL
      pt1, pt2 : integer_point;
    END_LOCAL;
    pt1 := startLine.dimensioned_figure.reference_line.start_point;
    pt2  := startLine.dimensioned_figure.reference_line.end_point;
    IF NOT colinear_points(pt, pt1, pt2) THEN
       RETURN(FALSE);
    ELSE
       REPEAT i := 1 TO SIZEOF(items);
         pt1 := items[i].end_line.dimensioned_figure.reference_line.start_point;
         pt2 := items[i].end_line.dimensioned_figure.reference_line.end_point;
         IF NOT colinear_points(pt, pt1, pt2) THEN
            RETURN(FALSE);
         END_IF;
       END_REPEAT;
    END_IF;
    RETURN(TRUE);
  END_FUNCTION;
