schematic_frame_model

Description:

The schematic_frame_model schema describes the structural connectivity and the implementation information of a frame in a schematic view. Different types of frame implementations are defined, including schematic_for_frame_implementation, schematic_if_frame_implementation and schematic_otherwise_frame_implementation. The structural connectivity information includes schematic_frame_bus and schematic_frame_net.

schematic_frame_implementation

  ENTITY schematic_frame_implementation
      ABSTRACT SUPERTYPE OF (ONEOF(schematic_for_frame_implementation,
                                   schematic_if_frame_implementation,
                                   schematic_otherwise_frame_implementation));
    busses                          : OPTIONAL SET [1:?] OF schematic_frame_bus;
    cell_property_displays          : OPTIONAL BAG [1:?] OF property_display;
    cluster_property_displays       : OPTIONAL BAG [1:?] OF property_display;
    comment_graphics_info           : OPTIONAL BAG [1:?] OF comment_graphics;
    global_port_representations     : OPTIONAL SET [1:?] OF
                                      schematic_global_port_representation; 
    implemented_frame               : frame;                      -- reference
    instance_implementations        : OPTIONAL SET [1:?] OF 
                                      schematic_instance_implementation; 
    master_port_representations     : OPTIONAL SET [1:?] OF
                                      schematic_master_port_representation; 
    master_port_bundle_representations : OPTIONAL SET [1:?] OF
                                      schematic_master_port_bundle_representation;
    nested_frame_implementations    : OPTIONAL SET [1:?] OF 
                                      schematic_frame_implementation;
    nets                            : OPTIONAL SET [1:?] OF schematic_frame_net;
    offpageconnectors               : OPTIONAL SET [1:?] OF
                                      schematic_offpageconnector; 
    onpageconnectors                : OPTIONAL SET [1:?] OF
                                      schematic_onpageconnector;  
    rippers                         : OPTIONAL SET [1:?] OF schematic_ripper;
    transform_position              : transform;
    view_property_displays          : OPTIONAL BAG [1:?] OF property_display;
  DERIVE
    containing_page : page -- reference
      := usedin_one_role(SELF, 
         'schematic_implementation_model.page.frame_implementations');
  UNIQUE
    valid_implemented_frame :
      (* A frame can only be implemented on a page once per schematic. *)
      implemented_frame;
  WHERE
    containment_constraint :
      (* A schematic_frame_implementation belongs to one page. *)
      belongs_to_one(SELF,
        'schematic_implementation_model.page.frame_implementations'); 

    valid_global_port_representations :
      (* The global_port_representations must only reference the global_ports
         or global_port_bundles which are defined in the containing edif. *)
      not_exists(QUERY(globalPortRepresentation <* global_port_representations |
        NOT ((globalPortRepresentation.referenced_global_port
             IN containing_page.containing_view.containing_cluster.containing_cell
                .containing_library.containing_edif.global_ports)
             OR
             (globalPortRepresentation.referenced_global_port
             IN containing_page.containing_view.containing_cluster.containing_cell
                .containing_library.containing_edif.global_port_bundles))));

    valid_master_port_representations :
      (* The master_port_representations must only reference the master_ports
         which are defined in the interface of the containing cluster. *) 
      not_exists(QUERY(portRepresentation <* master_port_representations |
        NOT (portRepresentation.implemented_master_port
             IN containing_page.containing_view.interface.ports)));

    valid_master_port_bundle_representations :
      (* The master_port_bundle_representations must only reference the
         master_port_bundles which are defined in the interface of the
         containing cluster. *)  
      not_exists(QUERY(portBundleRepresentation 
        <* master_port_bundle_representations |
        NOT (portBundleRepresentation.implemented_master_port_bundle
             IN containing_page.containing_view.interface.port_bundles)));

    valid_instance_implementations :
      (* The implemented instances are defined in the implemented_frame. *)
      not_exists(QUERY(implementation <* instance_implementations |
       NOT (implementation.implemented_instance IN implemented_frame.instances)));

    valid_nested_frame_implementations :
      (* Frames implemented in the nested_frame_implementations must be defined
         in the implemented_frame. *)
      not_exists(QUERY(nestedImplementation <* nested_frame_implementations |
        NOT (nestedImplementation.implemented_frame 
            IN implemented_frame.nested_frames)));  

    valid_cell_property_displays :
      (* The displayed properties are defined in the containing cell. *)
      not_exists(QUERY(cellPropertyDisplay <* cell_property_displays |
        NOT (cellPropertyDisplay.displayed_property
             IN containing_page.containing_view.containing_cluster
                .containing_cell.properties)));

    valid_cluster_property_displays :
      (* The displayed properties are defined in the containing
         internal_cluster. *) 
      not_exists(QUERY(clusterPropertyDisplay <* cluster_property_displays |
        NOT (clusterPropertyDisplay.displayed_property
             IN containing_page.containing_view.containing_cluster.properties)));

    valid_view_property_displays :
      (* The displayed properties are defined in the containing
         internal_schematic_view. *) 
      not_exists(QUERY(viewPropertyDisplay <* view_property_displays |
        NOT (viewPropertyDisplay.displayed_property
             IN containing_page.containing_view.properties)));
  END_ENTITY;

schematic_frame_generic_net

  ENTITY schematic_frame_generic_net
      ABSTRACT SUPERTYPE OF (ONEOF(schematic_frame_net, 
                                   schematic_frame_sub_net));
    associated_signal                   : frame_signal;         -- reference
    criticality                         : integer_token;                 
    delays                              : OPTIONAL SET [1:?] OF
                                          interconnect_delay; 
    document                            : OPTIONAL SET [1:?] OF documentation; 
    implementation                      : OPTIONAL
                                          schematic_interconnect_implementation; 
    joined_global_ports                 : OPTIONAL SET [1:?] OF -- reference
                                          global_port;
    joined_global_port_representations  : OPTIONAL SET [1:?] OF -- reference
                                          schematic_global_port_representation;
    joined_instance_ports               : OPTIONAL SET [1:?] OF -- reference
                                          instance_port;
    joined_instance_port_representations: OPTIONAL SET [1:?] OF
                                          schematic_instance_port_representation;
    joined_master_ports                 : OPTIONAL SET [1:?] OF -- reference
                                          master_port; 
    joined_master_port_representations  : OPTIONAL SET [1:?] OF -- reference
                                          schematic_master_port_representation; 
    joined_junctions                    : OPTIONAL SET [1:?] OF -- reference
                                          schematic_net_junction;
    joined_offpageconnectors            : OPTIONAL SET [1:?] OF -- reference
                                          schematic_offpageconnector;
    joined_onpageconnectors             : OPTIONAL SET [1:?] OF -- reference
                                          schematic_onpageconnector;
    joined_ripper_hotspots              : OPTIONAL SET [1:?] OF -- reference
                                          ripper_hotspot;
    joined_terminators                  : OPTIONAL SET [1:?] OF -- reference
                                          schematic_interconnect_terminator;
    name                                : OPTIONAL name_information;                 
    properties                          : OPTIONAL SET [1:?] OF property;
    sub_nets                            : OPTIONAL SET [1:?] OF
                                          schematic_frame_sub_net; 
  WHERE
    valid_sub_nets :
      (* No two schematic_frame_sub_nets may refer to the same set of ports and
         instantiatable schematic objects. *)

    not_exists(QUERY(net1 <* sub_nets |
      there_exists(QUERY(net2 <* sub_nets - net1 |
        (net1.joined_global_ports = net2.joined_global_ports) 
    AND (net1.joined_global_port_representations =
         net2.joined_global_port_representations)
    AND (net1.joined_instance_ports = net2.joined_instance_ports) 
    AND (net1.joined_instance_port_representations =
         net2.joined_instance_port_representations)
    AND (net1.joined_master_ports = net2.joined_master_ports) 
    AND (net1.joined_master_port_representations = 
         net2.joined_master_port_representations)
    AND (net1.joined_junctions = net2.joined_junctions)
    AND (net1.joined_offpageconnectors = net2.joined_offpageconnectors)
    AND (net1.joined_onpageconnectors = net2.joined_onpageconnectors)
    AND (net1.joined_ripper_hotspots = net2.joined_ripper_hotspots)
    AND (net1.joined_terminators = net2.joined_terminators)))));

    valid_global_port_representations :
      (* The joined global_port_representations reference global_ports which
         are joined by the interconnect. *)
      not_exists(QUERY(globalPortRepresentation
         <* joined_global_port_representations |
         NOT (globalPortRepresentation.referenced_global_port
              IN joined_global_ports)));

    valid_instance_port_representations :
      (* The joined instance_port_representations reference instance_ports
         which are joined by the interconnect. *)
      not_exists(QUERY(instancePortRepresentation
         <* joined_instance_port_representations |
         NOT (instancePortRepresentation.referenced_instance_port
              IN joined_instance_ports)));

    valid_master_port_representations :
      (* The joined master_port_representations reference master_ports which
         are joined by the interconnect. *)
      not_exists(QUERY(masterPortRepresentation
         <* joined_master_port_representations |
         NOT (masterPortRepresentation.implemented_master_port
              IN joined_master_ports)));

    valid_implementation :
      (* Implementation information may only be associated with a
         schematic_frame_generic_net when its sub_nets have no implementation
         information. *)
       not_defined;
  END_ENTITY;

Description:

 LEVEL 1
 =======
A schematic_frame_generic_net can be either a schematic_frame_net or a schematic_frame_sub_net. A schematic_frame_generic_net is used in the model because of the similarity between a schematic_frame_net and a schematic_frame_sub_net. A schematic_frame_net may be divided into schematic_frame_sub_nets and a schematic_frame_sub_net itself may also be divided into schematic_frame_sub_nets. No two schematic_frame_sub_nets at a given level may join to the same set of ports and instantiatable schematic objects.

Used by:

schematic_frame_sub_net schematic_net_junction

schematic_frame_net

  ENTITY schematic_frame_net
      SUBTYPE OF (schematic_frame_generic_net);
    junctions            : OPTIONAL SET [1:?] OF schematic_net_junction;
    schematic_wire_style : OPTIONAL schematic_wire_style;
    terminators          : OPTIONAL SET [1:?] OF
                           schematic_interconnect_terminator; 
  DERIVE
   containing_frame_implementation : schematic_frame_implementation -- reference
     := usedin_one_role(SELF,
        'schematic_frame_model.schematic_frame_implementation.nets'); 
  WHERE
    containment_constraint :
      (* A schematic_frame_net belongs to one schematic_frame_implementation. *)
      belongs_to_one(SELF, 
        'schematic_frame_model.schematic_frame_implementation.nets');

    valid_associated_signal :
      (* The frame_signal associated with a schematic_frame_net is defined in
         the implemented frame of the containing schematic_frame_implementation.*)
      associated_signal IN
        containing_frame_implementation.implemented_frame.frame_signals; 

    valid_joined_ports : 
      (* The ports joined by a schematic_frame_net must be a subset of the
         connectivity of the associated signal of the schematic_frame_net. *)

        (joined_global_ports <= associated_signal.joined_global_ports)
    AND (joined_instance_ports <= associated_signal.joined_instance_ports)
    AND (joined_master_ports <= associated_signal.joined_master_ports);

    valid_joined_instance_ports :
      (* The joined instance_ports are defined in instances which are
         implemented in the containing schematic_frame_implementation. *)
      not_exists(QUERY(instancePort <* joined_instance_ports |
        NOT (usedin_one_role(instancePort.referenced_instance,
                             specified_role('schematic_implementation_model',
                                            'schematic_instance_implementation',
                                            'implemented_instance'))
             IN containing_frame_implementation.instance_implementations)));

    valid_joined_instantiatable_objects :
      (* All joined schematic_global_port_representations,
         schematic_master_port_representations, schematic_offpageconnectors,
         schematic_onpageconnectors and ripper_hotspots are defined in
         the containing schematic_frame_implementation. *)

        (joined_global_port_representations
         <= containing_frame_implementation.global_port_representations)
    AND (joined_master_port_representations
         <= containing_frame_implementation.master_port_representations)
    AND (joined_offpageconnectors
         <= containing_frame_implementation.offpageconnectors)
    AND (joined_onpageconnectors
         <= containing_frame_implementation.onpageconnectors)
    AND (not_exists(QUERY(hotspot <* joined_ripper_hotspots |
        NOT (hotspot.containing_ripper 
             IN containing_frame_implementation.rippers))));

    valid_joined_junctions :
      (* The joined junctions must be defined in the schematic_frame_net. *)
      joined_junctions <= junctions;

    valid_joined_terminators :
      (* The joined terminators must be defined in the schematic_frame_net. *)
      joined_terminators <= terminators;
  END_ENTITY;

Description:

 LEVEL 1
 =======
A schematic_frame_net is defined within a schematic_frame_implementation which describes the structural connectivity information of a frame in a schematic view. Its associated frame_signal must be defined in the implemented frame of the containing schematic_frame_implementation. The ports joined by a schematic_frame_net must be a subset of the ones joined by its associated frame_signal. Also, all joined instantiatable schematic objects must be defined in the containing schematic_frame_implementation.

Used by:

schematic_frame_implementation

schematic_frame_sub_net

  ENTITY schematic_frame_sub_net
      SUBTYPE OF (schematic_frame_generic_net);
  DERIVE
    containing_generic_net : schematic_frame_generic_net           -- reference
      := usedin_one_role(SELF,
         'schematic_frame_model.schematic_frame_sub_net.sub_nets'); 
  WHERE
    containment_constraint :
      (* A schematic_frame_sub_net belongs to one schematic_frame_generic_net. *) 
      belongs_to_one(SELF,
        'schematic_frame_model.schematic_frame_generic_net.sub_nets');

    valid_associated_signal :
      (* The frame_signal associated with a schematic_frame_sub_net is the same
         as that of its containing schematic_frame_generic_net. *)
      associated_signal :=: containing_generic_net.associated_signal;

    valid_joined_ports :
      (* The ports joined by a schematic_frame_sub_net must be a subset of the
         ports joined by its immediately containing
         schematic_frame_generic_net. *)  

        (joined_global_ports <= containing_generic_net.joined_global_ports)
    AND (joined_instance_ports <= containing_generic_net.joined_instance_ports)
    AND (joined_master_ports <= containing_generic_net.joined_master_ports);

    valid_joined_instantiatable_objects :
      (* The schematic_global_port_representations,
         schematic_instance_port_representations,
         schematic_master_port_representations, schematic_net_junctions,
         schematic_offpageconnectors, schematic_onpageconnectors and
         ripper_hotspots joined by a schematic_frame_sub_net must be a subset
         of the ones joined by its immediately containing
         schematic_frame_generic_net. *)

        (joined_global_port_representations 
         <= containing_generic_net.joined_global_port_representations)
    AND (joined_junctions <= containing_generic_net.joined_junctions)
    AND (joined_instance_port_representations 
         <= containing_generic_net.joined_instance_port_representations)
    AND (joined_master_port_representations 
         <= containing_generic_net.joined_master_port_representations)
    AND (joined_offpageconnectors 
         <= containing_generic_net.joined_offpageconnectors)
    AND (joined_onpageconnectors 
         <= containing_generic_net.joined_onpageconnectors)
    AND (joined_ripper_hotspots <= containing_generic_net.joined_ripper_hotspots);

    valid_joined_terminators :
      (* The joined terminators must be defined in the containing
         schematic_frame_net. *) 
      not_defined;
  END_ENTITY;

schematic_frame_generic_bus

  ENTITY schematic_frame_generic_bus
      ABSTRACT SUPERTYPE OF (ONEOF(schematic_frame_bus, 
                                   schematic_frame_bus_slice, 
                                   schematic_frame_sub_bus));
    associated_signal_group             : frame_signal_group;   -- reference
    bus_slices                          : OPTIONAL SET [1:?] OF
                                          schematic_frame_bus_slice; 
    criticality                         : integer_token;                  
    delays                              : OPTIONAL SET [1:?] OF
                                          interconnect_delay; 
    document                            : OPTIONAL SET [1:?] OF documentation;
    implementation                      : OPTIONAL
                                          schematic_interconnect_implementation; 
    joined_global_ports                 : OPTIONAL SET [1:?] OF -- reference
                                          global_port;
    joined_global_port_bundles          : OPTIONAL SET [1:?] OF -- reference
                                          global_port_bundle;
    joined_global_port_representations  : OPTIONAL SET [1:?] OF -- reference
                                          schematic_global_port_representation;
    joined_instance_ports               : OPTIONAL SET [1:?] OF -- reference
                                          instance_port;
    joined_instance_port_bundles        : OPTIONAL SET [1:?] OF -- reference
                                          instance_port_bundle;
    joined_instance_port_representations: OPTIONAL SET [1:?] OF -- reference
                                          schematic_instance_port_representation;
    joined_instance_port_bundle_representations: OPTIONAL SET [1:?] OF-- reference
                                    schematic_instance_port_bundle_representation;
    joined_local_master_port_groups     : OPTIONAL SET [1:?] OF -- reference
                                          local_master_port_group;
    joined_master_ports                 : OPTIONAL SET [1:?] OF -- reference
                                          master_port; 
    joined_master_port_bundles          : OPTIONAL SET [1:?] OF -- reference
                                          master_port_bundle;           
    joined_master_port_representations  : OPTIONAL SET [1:?] OF -- reference
                                          schematic_master_port_representation; 
    joined_master_port_bundle_representations : OPTIONAL SET [1:?] OF -- reference
                                    schematic_master_port_bundle_representation;
    joined_junctions                    : OPTIONAL SET [1:?] OF -- reference
                                          schematic_bus_junction;
    joined_offpageconnectors            : OPTIONAL SET [1:?] OF -- reference
                                          schematic_offpageconnector;
    joined_onpageconnectors             : OPTIONAL SET [1:?] OF -- reference
                                          schematic_onpageconnector;
    joined_ripper_hotspots              : OPTIONAL SET [1:?] OF -- reference
                                          ripper_hotspot;
    joined_terminators                  : OPTIONAL SET [1:?] OF -- reference
                                          schematic_interconnect_terminator;
    name                                : OPTIONAL name_information;                 
    properties                          : OPTIONAL SET [1:?] OF property;
    sub_busses                          : OPTIONAL SET [1:?] OF
                                          schematic_frame_sub_bus; 
  DERIVE
    signal_group_size : INTEGER 
      := SIZEOF(associated_signal_group.flattened_signal_list);
  WHERE
    valid_joined_port_size :
      (* All joined global_ports, global_port_bundles,
         local_master_port_groups, master_ports and master_port_bundles must 
         have the same port size as the size of the flattened signal_group. *)

        (not_exists(QUERY(joinedGlobalPort <* joined_global_ports |
           joinedGlobalPort.size <> signal_group_size)))
    AND (not_exists(QUERY(joinedGlobalPortBundle <* joined_global_port_bundles |
           joinedGlobalPortBundle.size <> signal_group_size)))
    AND (not_exists(QUERY(joinedLocalMasterPortGroup 
           <* joined_local_master_port_groups |
           joinedLocalMasterPortGroup.size <> signal_group_size)))
    AND (not_exists(QUERY(joinedMasterPort <* joined_master_ports |
           joinedMasterPort.size <> signal_group_size)))
    AND (not_exists(QUERY(joinedMasterPortBundle <* joined_master_port_bundles |
           joinedMasterPortBundle.size <> signal_group_size)));

    valid_structural_connectivity :
      not_defined;

    valid_bus_slices :
      (* No two bus_slices within a schematic_frame_generic_bus may reference
         the same frame_signal_group. *)
      not_exists(QUERY(busSlice1 <* bus_slices |
        there_exists(QUERY(busSlice2 <* bus_slices - busSlice1 |
          busSlice1.associated_signal_group :=: 
            busSlice2.associated_signal_group))));

    valid_sub_busses :
      (* No two sub_busses within a schematic_generic_bus may refer to the same
         set of ports and instantiatable schematic objects. *)

    not_exists(QUERY(bus1 <* sub_busses |
      there_exists(QUERY(bus2 <* sub_busses - bus1 |
        (bus1.joined_global_ports = bus2.joined_global_ports) 
    AND (bus1.joined_global_port_bundles = bus2.joined_global_port_bundles) 
    AND (bus1.joined_global_port_representations
         = bus2.joined_global_port_representations)
    AND (bus1.joined_instance_ports = bus2.joined_instance_ports) 
    AND (bus1.joined_instance_port_bundles = bus2.joined_instance_port_bundles) 
    AND (bus1.joined_instance_port_representations
         = bus2.joined_instance_port_representations)
    AND (bus1.joined_instance_port_bundle_representations
         = bus2.joined_instance_port_bundle_representations)
    AND (bus1.joined_local_master_port_groups 
         = bus2.joined_local_master_port_groups) 
    AND (bus1.joined_master_ports = bus2.joined_master_ports) 
    AND (bus1.joined_master_port_bundles = bus2.joined_master_port_bundles) 
    AND (bus1.joined_master_port_representations 
         = bus2.joined_master_port_representations)
    AND (bus1.joined_master_port_bundle_representations 
         = bus2.joined_master_port_bundle_representations)
    AND (bus1.joined_junctions = bus2.joined_junctions)
    AND (bus1.joined_offpageconnectors = bus2.joined_offpageconnectors)
    AND (bus1.joined_onpageconnectors = bus2.joined_onpageconnectors)
    AND (bus1.joined_ripper_hotspots = bus2.joined_ripper_hotspots)
    AND (bus1.joined_terminators = bus2.joined_terminators)))));

    valid_global_port_representations :
      (* The joined global_port_representations reference global_ports or
         global_port_bundles which are joined by the interconnect. *)
      not_exists(QUERY(globalPortRepresentation
         <* joined_global_port_representations |
         NOT ((globalPortRepresentation.referenced_global_port
               IN joined_global_ports)
           OR (globalPortRepresentation.referenced_global_port
               IN joined_global_port_bundles))));

    valid_instance_port_representations :
      (* The joined instance_port_representations reference instance_ports
         which are joined by the interconnect. *)
      not_exists(QUERY(instancePortRepresentation
         <* joined_instance_port_representations |
         NOT (instancePortRepresentation.referenced_instance_port
              IN joined_instance_ports)));

    valid_instance_port_bundle_representations :
      (* The joined instance_port_bundle_representations reference
         instance_port_bundles which are joined by the interconnect. *)
      not_exists(QUERY(instancePortBundleRepresentation
         <* joined_instance_port_bundle_representations |
         NOT (instancePortBundleRepresentation.referenced_instance_port_bundle
              IN joined_instance_port_bundles)));

    valid_master_port_representations :
      (* The joined master_port_representations reference master_ports which
         are joined by the interconnect. *) 
      not_exists(QUERY(masterPortRepresentation
         <* joined_master_port_representations |
         NOT (masterPortRepresentation.implemented_master_port
              IN joined_master_ports)));

    valid_master_port_bundle_representations :
      (* The joined master_port_bundle_representations reference
         master_port_bundles which are joined by the interconnect. *) 
      not_exists(QUERY(masterPortBundleRepresentation
         <* joined_master_port_bundle_representations |
         NOT (masterPortBundleRepresentation.implemented_master_port_bundle
              IN joined_master_port_bundles)));

    valid_implementation :
      (* Implementation information may only be associated with a
         schematic_frame_generic_bus when its sub_busses have no implementation
         information. *)
       not_defined;
  END_ENTITY;

Description:

 LEVEL 1
 =======
A schematic_frame_generic_bus can be either a schematic_frame_bus, a schematic_frame_bus_slice or a schematic_frame_sub_bus. A schematic_frame_generic_bus is used in the model because of the similarity between a schematic_frame_bus, a schematic_frame_bus_slice and a schematic_frame_sub_bus. They may all contain schematic_frame_bus_slices and schematic_frame_sub_busses. No two schematic_frame_bus_slices at a given level may be associated with the same frame_signal_group. No two schematic_frame_sub_busses at a given level may join to the same set of ports and instantiatable schematic objects.

schematic_frame_bus

  ENTITY schematic_frame_bus
      SUBTYPE OF (schematic_frame_generic_bus);
    junctions            : OPTIONAL SET [1:?] OF schematic_bus_junction;
    schematic_wire_style : OPTIONAL schematic_wire_style;
    terminators          : OPTIONAL SET [1:?] OF
                           schematic_interconnect_terminator; 
  DERIVE
    containing_frame_implementation : schematic_frame_implementation -- reference
      := usedin_one_role(SELF,
         'schematic_frame_model.schematic_frame_implementation.busses');
  WHERE
    containment_constraint :
      (* A schematic_frame_bus belongs to one schematic_frame_implementation. *) 
      belongs_to_one(SELF,
        'schematic_frame_model.schematic_frame_implementation.busses');

    valid_associated_signal_group :
      (* The frame_signal_group associated with a schematic_frame_bus is
         defined in the frame which is implemented. *)
      associated_signal_group IN
        containing_frame_implementation.implemented_frame.frame_signal_groups; 

    valid_joined_instance_ports :
      (* The joined instance_ports are defined in instances which are
         implemented in the containing schematic_frame_implementation. *)
      not_exists(QUERY(instancePort <* joined_instance_ports |
        NOT (usedin_one_role(instancePort.referenced_instance,
                             specified_role('schematic_implementation_model',
                                            'schematic_instance_implementation',
                                            'implemented_instance'))
             IN containing_frame_implementation.instance_implementations)));

    valid_joined_instance_port_bundles :
      (* The joined instance_port_bundles are defined in instances which are
         implemented in the containing schematic_frame_implementation. *)
      not_exists(QUERY(instancePortBundle <* joined_instance_ports |
        NOT (usedin_one_role(instancePortBundle.referenced_instance,
                             specified_role('schematic_implementation_model',
                                            'schematic_instance_implementation',
                                            'implemented_instance'))
             IN containing_frame_implementation.instance_implementations)));

    valid_joined_instantiatable_objects :
      (* All joined schematic_global_port_representations,
         schematic_master_port_representations,
         schematic_master_port_bundle_representations,
         schematic_offpageconnectors, schematic_onpageconnectors and
         ripper_hotspots are defined in the containing
         schematic_frame_implementation. *) 

        (joined_global_port_representations
         <= containing_frame_implementation.global_port_representations)
    AND (joined_master_port_representations 
         <= containing_frame_implementation.master_port_representations)
    AND (joined_master_port_bundle_representations 
         <= containing_frame_implementation.master_port_bundle_representations)
    AND (joined_offpageconnectors 
         <= containing_frame_implementation.offpageconnectors)
    AND (joined_onpageconnectors 
         <= containing_frame_implementation.onpageconnectors)
    AND (not_exists(QUERY(hotspot <* joined_ripper_hotspots |
        NOT (hotspot.containing_ripper 
             IN containing_frame_implementation.rippers))));

    valid_joined_junctions :
      (* The joined junctions must be defined in the schematic_frame_bus. *)
      joined_junctions <= junctions;

    valid_joined_terminators :
      (* The joined terminators must be defined in the schematic_frame_bus. *)
      joined_terminators <= terminators;
  END_ENTITY;

  ENTITY schematic_frame_sub_bus
      SUBTYPE OF (schematic_frame_generic_bus);
  DERIVE
   containing_generic_bus : schematic_frame_generic_bus -- reference
     := usedin_one_role(SELF, 
        'schematic_frame_model.schematic_frame_generic_bus.sub_busses');
  WHERE
    containment_constraint :
      (* A schematic_frame_sub_bus belongs to one schematic_frame_generic_bus. *) 
      belongs_to_one(SELF,
        'schematic_frame_model.schematic_frame_generic_bus.sub_busses');

    valid_associated_signal_group :
      (* The frame_signal_group associated with a schematic_frame_sub_bus is
         the same as its containing schematic_frame_generic_bus. *)
      associated_signal_group :=: containing_generic_bus.associated_signal_group;

    valid_joined_ports :
      (* The ports joined by a schematic_frame_sub_bus must be a subset of the
         ports joined by its immediately containing
         schematic_frame_generic_bus. *) 

        (joined_global_ports <= containing_generic_bus.joined_global_ports)
    AND (joined_global_port_bundles 
         <= containing_generic_bus.joined_global_port_bundles)
    AND (joined_instance_ports <= containing_generic_bus.joined_instance_ports)
    AND (joined_instance_port_bundles 
         <= containing_generic_bus.joined_instance_port_bundles)
    AND (joined_local_master_port_groups 
         <= containing_generic_bus.joined_local_master_port_groups)
    AND (joined_master_ports <= containing_generic_bus.joined_master_ports)
    AND (joined_master_port_bundles 
         <= containing_generic_bus.joined_master_port_bundles);

    valid_joined_instantiatable_objects :
      (* The schematic_global_port_representations,
         schematic_master_port_representations,
         schematic_master_port_bundle_representations,
         schematic_offpageconnectors, schematic_onpageconnectors and
         ripper_hotspots joined by a schematic_frame_sub_bus must be a subset
         of the ones joined by its immediately containing
         schematic_frame_generic_bus. *)   

        (joined_global_port_representations 
         <= containing_generic_bus.joined_global_port_representations)
    AND (joined_instance_port_representations 
         <= containing_generic_bus.joined_instance_port_representations)
    AND (joined_instance_port_bundle_representations 
         <= containing_generic_bus.joined_instance_port_bundle_representations)
    AND (joined_master_port_representations 
         <= containing_generic_bus.joined_master_port_representations)
    AND (joined_master_port_bundle_representations 
         <= containing_generic_bus.joined_master_port_bundle_representations)
    AND (joined_offpageconnectors 
         <= containing_generic_bus.joined_offpageconnectors) 
    AND (joined_onpageconnectors 
         <= containing_generic_bus.joined_onpageconnectors)
    AND (joined_junctions <= containing_generic_bus.joined_junctions);

    valid_joined_terminators :
      (* The joined terminators must be defined in the containing
         schematic_frame_bus. *) 
      not_defined;
  END_ENTITY;
